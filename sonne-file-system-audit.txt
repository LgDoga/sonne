use std::{
    fs,
    path::{Path, PathBuf},
    time::Duration,
};

use rusqlite::{Connection, OpenFlags};
use schema::{INITIAL_SCHEMA, SCHEMA_VERSION};

pub struct Storage {
    connection: Connection,
    root_dir: PathBuf,
}

impl Storage {
    pub fn initialize(app_data_dir: &Path) -> Result<Self, String> {
        let root_dir = app_data_dir.join("storage");

        for directory in [
            root_dir.join("blobs"),
            root_dir.join("temp"),
            root_dir.join("extracted"),
            root_dir.join("indexes"),
        ] {
            fs::create_dir_all(&directory)
                .map_err(|error| format!("Cannot create storage directory: {error}"))?;
        }

        let database_path = root_dir.join("sonne.db");

        let connection = Connection::open_with_flags(
            &database_path,
            OpenFlags::SQLITE_OPEN_READ_WRITE
                | OpenFlags::SQLITE_OPEN_CREATE
                | OpenFlags::SQLITE_OPEN_FULL_MUTEX,
        )
        .map_err(|error| format!("Cannot open Sonne database: {error}"))?;

        connection
            .busy_timeout(Duration::from_secs(10))
            .map_err(|error| format!("Cannot configure database timeout: {error}"))?;

        connection
            .execute_batch(
                "
                PRAGMA foreign_keys = ON;
                PRAGMA journal_mode = WAL;
                PRAGMA synchronous = NORMAL;
                PRAGMA temp_store = MEMORY;
                PRAGMA mmap_size = 268435456;
                ",
            )
            .map_err(|error| format!("Cannot configure SQLite: {error}"))?;

        let current_version: i64 = connection
            .pragma_query_value(None, "user_version", |row| row.get(0))
            .map_err(|error| format!("Cannot read database version: {error}"))?;

        if current_version > SCHEMA_VERSION {
            return Err(format!(
                "Database version {current_version} is newer than supported version {SCHEMA_VERSION}"
            ));
        }

        if current_version < SCHEMA_VERSION {
            connection
                .execute_batch(INITIAL_SCHEMA)
                .map_err(|error| format!("Cannot initialize database schema: {error}"))?;

            connection
                .pragma_update(None, "user_version", SCHEMA_VERSION)
                .map_err(|error| format!("Cannot update database version: {error}"))?;
        }

        Ok(Self {
            connection,
            root_dir,
        })
    }

    pub fn connection(&self) -> &Connection {
        &self.connection
    }

    pub fn root_dir(&self) -> &Path {
        &self.root_dir
    }
}

use std::{
    fs::{self, File, OpenOptions},
    io::{Read, Write},
    path::{Component, Path, PathBuf},
    time::{SystemTime, UNIX_EPOCH},
};

use rusqlite::{OptionalExtension, params};
use serde::Serialize;
use sha2::{Digest, Sha256};
use uuid::Uuid;

use super::Storage;

const BUFFER_SIZE: usize = 1024 * 1024;

#[derive(Debug, Clone, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct BlobRecord {
    pub sha256: String,
    pub relative_path: String,
    pub original_size: i64,
    pub stored_size: i64,
}

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))
}

fn digest_to_hex(digest: &[u8]) -> String {
    digest.iter().map(|byte| format!("{byte:02x}")).collect()
}

fn hash_file(path: &Path) -> Result<String, String> {
    let mut file =
        File::open(path).map_err(|error| format!("Cannot open existing blob: {error}"))?;

    let mut hasher = Sha256::new();
    let mut buffer = vec![0u8; BUFFER_SIZE];

    loop {
        let bytes_read = file
            .read(&mut buffer)
            .map_err(|error| format!("Cannot verify existing blob: {error}"))?;

        if bytes_read == 0 {
            break;
        }

        hasher.update(&buffer[..bytes_read]);
    }

    Ok(digest_to_hex(&hasher.finalize()))
}

fn resolve_relative_path(root: &Path, relative: &str) -> Result<PathBuf, String> {
    let path = Path::new(relative);

    if path.is_absolute()
        || !path
            .components()
            .all(|component| matches!(component, Component::Normal(_)))
        || path.components().next() != Some(Component::Normal("blobs".as_ref()))
    {
        return Err("Invalid internal blob path".to_string());
    }

    Ok(root.join(path))
}

impl Storage {
    pub fn resolve_blob_path(&self, relative: &str) -> Result<PathBuf, String> {
        let candidate = resolve_relative_path(&self.root_dir, relative)?;

        let storage_root = self
            .root_dir
            .canonicalize()
            .map_err(|error| format!("Cannot resolve storage directory: {error}"))?;

        let blobs_root = self
            .root_dir
            .join("blobs")
            .canonicalize()
            .map_err(|error| format!("Cannot resolve blob storage directory: {error}"))?;

        if !blobs_root.starts_with(&storage_root) {
            return Err("Blob storage directory escapes storage root".to_string());
        }

        let canonical = candidate
            .canonicalize()
            .map_err(|error| format!("Cannot resolve blob file: {error}"))?;

        if !canonical.starts_with(&blobs_root) {
            return Err("Blob path escapes storage directory".to_string());
        }

        let metadata = fs::symlink_metadata(&candidate)
            .map_err(|error| format!("Cannot inspect blob file: {error}"))?;

        if metadata.file_type().is_symlink() || !metadata.is_file() {
            return Err("Blob path is not a regular file".to_string());
        }

        Ok(canonical)
    }

    pub fn store_blob<R: Read>(&mut self, mut reader: R) -> Result<BlobRecord, String> {
        let operation_id = Uuid::new_v4().to_string();
        let timestamp = now_timestamp()?;

        let temp_relative = format!("temp/{operation_id}.part");
        let temp_path = self.root_dir.join(&temp_relative);

        self.connection
            .execute(
                "INSERT INTO storage_operations (
                    id, operation_type, state, created_at, updated_at
                 ) VALUES (?1, 'store_blob', 'PENDING', ?2, ?2)",
                params![operation_id, timestamp],
            )
            .map_err(|error| format!("Cannot create storage journal entry: {error}"))?;

        self.connection
            .execute(
                "UPDATE storage_operations
                 SET state = 'PROCESSING', source_path = ?1, updated_at = ?2
                 WHERE id = ?3",
                params![temp_relative, timestamp, operation_id],
            )
            .map_err(|error| format!("Cannot update storage journal: {error}"))?;

        let mut output = OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(&temp_path)
            .map_err(|error| format!("Cannot create temporary blob: {error}"))?;

        let mut hasher = Sha256::new();
        let mut buffer = vec![0u8; BUFFER_SIZE];
        let mut total_size: u64 = 0;

        loop {
            let bytes_read = reader
                .read(&mut buffer)
                .map_err(|error| format!("Cannot read incoming file: {error}"))?;

            if bytes_read == 0 {
                break;
            }

            output
                .write_all(&buffer[..bytes_read])
                .map_err(|error| format!("Cannot write temporary blob: {error}"))?;

            hasher.update(&buffer[..bytes_read]);

            total_size = total_size
                .checked_add(bytes_read as u64)
                .ok_or_else(|| "File size overflow".to_string())?;
        }

        output
            .sync_all()
            .map_err(|error| format!("Cannot synchronize temporary blob: {error}"))?;

        drop(output);

        let original_size = i64::try_from(total_size)
            .map_err(|error| format!("File is too large for SQLite metadata: {error}"))?;

        let sha256 = digest_to_hex(&hasher.finalize());

        let existing: Option<(String, i64)> = self
            .connection
            .query_row(
                "SELECT relative_path, original_size
                 FROM blobs WHERE sha256 = ?1",
                params![sha256],
                |row| Ok((row.get(0)?, row.get(1)?)),
            )
            .optional()
            .map_err(|error| format!("Cannot check blob deduplication: {error}"))?;

        let relative_path = if let Some((existing_relative, _)) = existing {
            let existing_path = resolve_relative_path(&self.root_dir, &existing_relative)?;

            let existing_is_valid = existing_path.is_file()
                && hash_file(&existing_path)
                    .map(|hash| hash == sha256)
                    .unwrap_or(false);

            if existing_is_valid {
                fs::remove_file(&temp_path)
                    .map_err(|error| format!("Cannot remove duplicate temporary file: {error}"))?;

                self.connection
                    .execute(
                        "UPDATE blobs SET last_accessed_at = ?1 WHERE sha256 = ?2",
                        params![now_timestamp()?, sha256],
                    )
                    .map_err(|error| format!("Cannot update blob access time: {error}"))?;

                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET blob_sha256 = ?1, state = 'COMPLETED', updated_at = ?2
                         WHERE id = ?3",
                        params![sha256, now_timestamp()?, operation_id],
                    )
                    .map_err(|error| format!("Cannot complete deduplication journal: {error}"))?;

                return Ok(BlobRecord {
                    sha256,
                    relative_path: existing_relative,
                    original_size,
                    stored_size: original_size,
                });
            }

            if let Some(parent) = existing_path.parent() {
                fs::create_dir_all(parent)
                    .map_err(|error| format!("Cannot recreate blob directory: {error}"))?;
            }

            self.connection
                .execute(
                    "UPDATE storage_operations
                     SET state = 'COMMITTING', target_path = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![existing_relative, now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot update repair journal: {error}"))?;

            fs::rename(&temp_path, &existing_path)
                .map_err(|error| format!("Cannot repair existing blob: {error}"))?;

            self.connection
                .execute(
                    "UPDATE blobs
                     SET original_size = ?1, stored_size = ?1,
                         storage_state = 'original',
                         compression_format = NULL,
                         last_accessed_at = ?2,
                         verified_at = ?2
                     WHERE sha256 = ?3",
                    params![original_size, now_timestamp()?, sha256],
                )
                .map_err(|error| format!("Cannot update repaired blob metadata: {error}"))?;

            existing_relative
        } else {
            let prefix = &sha256[..2];
            let relative = format!("blobs/{prefix}/{sha256}.blob");
            let target_path = resolve_relative_path(&self.root_dir, &relative)?;

            if let Some(parent) = target_path.parent() {
                fs::create_dir_all(parent)
                    .map_err(|error| format!("Cannot create blob directory: {error}"))?;
            }

            self.connection
                .execute(
                    "UPDATE storage_operations
                     SET state = 'VERIFYING', target_path = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![relative, now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot update verification journal: {error}"))?;

            let orphan_is_valid = target_path.is_file()
                && hash_file(&target_path)
                    .map(|hash| hash == sha256)
                    .unwrap_or(false);

            if orphan_is_valid {
                fs::remove_file(&temp_path)
                    .map_err(|error| format!("Cannot remove duplicate orphan blob: {error}"))?;
            } else {
                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET state = 'COMMITTING', updated_at = ?1
                         WHERE id = ?2",
                        params![now_timestamp()?, operation_id],
                    )
                    .map_err(|error| format!("Cannot update commit journal: {error}"))?;

                fs::rename(&temp_path, &target_path)
                    .map_err(|error| format!("Cannot commit blob file: {error}"))?;
            }

            relative
        };

        let timestamp = now_timestamp()?;

        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start blob metadata transaction: {error}"))?;

        transaction
            .execute(
                "INSERT INTO blobs (
                    sha256, relative_path, original_size, stored_size,
                    storage_state, created_at, last_accessed_at, verified_at
                 ) VALUES (?1, ?2, ?3, ?3, 'original', ?4, ?4, ?4)
                 ON CONFLICT(sha256) DO UPDATE SET
                    relative_path = excluded.relative_path,
                    original_size = excluded.original_size,
                    stored_size = excluded.stored_size,
                    storage_state = 'original',
                    compression_format = NULL,
                    last_accessed_at = excluded.last_accessed_at,
                    verified_at = excluded.verified_at",
                params![sha256, relative_path, original_size, timestamp],
            )
            .map_err(|error| format!("Cannot save blob metadata: {error}"))?;

        transaction
            .execute(
                "UPDATE storage_operations
                 SET blob_sha256 = ?1, state = 'COMPLETED', updated_at = ?2
                 WHERE id = ?3",
                params![sha256, timestamp, operation_id],
            )
            .map_err(|error| format!("Cannot complete blob journal: {error}"))?;

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit blob metadata: {error}"))?;

        Ok(BlobRecord {
            sha256,
            relative_path,
            original_size,
            stored_size: original_size,
        })
    }
}

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))
}

fn file_metadata(path: &Path) -> Result<(String, String, String), String> {
    let file_name = path
        .file_name()
        .ok_or_else(|| "Selected file has no filename".to_string())?
        .to_string_lossy()
        .into_owned();

    let extension = path
        .extension()
        .and_then(|value| value.to_str())
        .unwrap_or("")
        .to_ascii_lowercase();

    let mime_type = match extension.as_str() {
        "pdf" => "application/pdf",
        "png" => "image/png",
        "jpg" | "jpeg" => "image/jpeg",
        "webp" => "image/webp",
        "gif" => "image/gif",
        "bmp" => "image/bmp",
        "tif" | "tiff" => "image/tiff",
        "avif" => "image/avif",
        "heic" | "heif" => "image/heic",
        "txt" | "log" | "csv" | "tsv" => "text/plain",
        "md" | "markdown" => "text/markdown",
        "html" | "htm" => "text/html",
        "json" => "application/json",
        "xml" => "application/xml",
        "js" | "mjs" | "cjs" => "text/javascript",
        "ts" | "tsx" => "text/typescript",
        "css" => "text/css",
        "rs" => "text/rust",
        "py" => "text/x-python",
        "java" => "text/x-java-source",
        "c" | "h" => "text/x-c",
        "cpp" | "hpp" | "cc" => "text/x-c++",
        "sh" | "bash" => "text/x-shellscript",
        "yaml" | "yml" => "application/yaml",
        "toml" => "application/toml",
        "docx" => "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
        "pptx" => "application/vnd.openxmlformats-officedocument.presentationml.presentation",
        "xlsx" => "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        "odt" => "application/vnd.oasis.opendocument.text",
        "ods" => "application/vnd.oasis.opendocument.spreadsheet",
        "odp" => "application/vnd.oasis.opendocument.presentation",
        "epub" => "application/epub+zip",
        "zip" => "application/zip",
        "gz" => "application/gzip",
        "tar" => "application/x-tar",
        "rtf" => "application/rtf",
        _ => "application/octet-stream",
    };

    Ok((file_name, extension, mime_type.to_string()))
}

struct PreparedAttachment {
    blob_sha256: String,
    file_name: String,
    extension: String,
    mime_type: String,
}

impl Storage {
    fn import_selected_paths(
        &mut self,
        chat_id: &str,
        paths: Vec<PathBuf>,
    ) -> Result<Vec<AttachmentReferenceRecord>, String> {
        if paths.is_empty() {
            return self.list_composer_attachments(chat_id);
        }

        let timestamp = now_timestamp()?;
        let operation_id = Uuid::new_v4().to_string();

        let chat_exists: bool = self
            .connection
            .query_row(
                "SELECT EXISTS(SELECT 1 FROM chats WHERE id = ?1)",
                params![chat_id],
                |row| row.get(0),
            )
            .map_err(|error| format!("Cannot verify import chat: {error}"))?;

        if !chat_exists {
            return Err(format!("Chat not found: {chat_id}"));
        }

        let existing = self.list_composer_attachments(chat_id)?;
        let mut position = i64::try_from(existing.len())
            .map_err(|error| format!("Attachment position overflow: {error}"))?;

        self.connection
            .execute(
                "INSERT INTO storage_operations (
                    id, operation_type, state, metadata_json,
                    created_at, updated_at
                ) VALUES (?1, 'import_batch', 'PENDING', ?2, ?3, ?3)",
                params![
                    operation_id,
                    json!({
                        "chatId": chat_id,
                        "files": []
                    })
                    .to_string(),
                    timestamp
                ],
            )
            .map_err(|error| format!("Cannot create import journal: {error}"))?;

        self.connection
            .execute(
                "UPDATE storage_operations
                 SET state = 'PROCESSING', updated_at = ?1
                 WHERE id = ?2",
                params![now_timestamp()?, operation_id],
            )
            .map_err(|error| format!("Cannot start import journal: {error}"))?;

        let preparation = (|| -> Result<Vec<PreparedAttachment>, String> {
            let mut prepared = Vec::with_capacity(paths.len());

            for path in paths {
                let metadata = std::fs::metadata(&path)
                    .map_err(|error| format!("Cannot inspect file {}: {error}", path.display()))?;

                if !metadata.is_file() {
                    return Err(format!("Not a regular file: {}", path.display()));
                }

                let (file_name, extension, mime_type) = file_metadata(&path)?;

                let file = File::open(&path)
                    .map_err(|error| format!("Cannot open {file_name}: {error}"))?;

                let blob = self.store_blob(file)?;

                prepared.push(PreparedAttachment {
                    blob_sha256: blob.sha256,
                    file_name,
                    extension,
                    mime_type,
                });

                let files_json: Vec<_> = prepared
                    .iter()
                    .map(|item| {
                        json!({
                            "blobSha256": item.blob_sha256,
                            "fileName": item.file_name,
                            "extension": item.extension,
                            "mimeType": item.mime_type
                        })
                    })
                    .collect();

                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET metadata_json = ?1, updated_at = ?2
                         WHERE id = ?3",
                        params![
                            json!({
                                "chatId": chat_id,
                                "files": files_json
                            })
                            .to_string(),
                            now_timestamp()?,
                            operation_id
                        ],
                    )
                    .map_err(|error| format!("Cannot update import journal: {error}"))?;
            }

            Ok(prepared)
        })();

        let prepared = match preparation {
            Ok(prepared) => prepared,
            Err(error) => {
                let _ = self.connection.execute(
                    "UPDATE storage_operations
                     SET state = 'FAILED', error_message = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![error, now_timestamp().unwrap_or(timestamp), operation_id],
                );
                return Err(error);
            }
        };

        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start atomic attachment transaction: {error}"))?;

        let commit_result = (|| -> Result<(), String> {
            for item in &prepared {
                let version = super::attachments::create_attachment_version_in_transaction(
                    &transaction,
                    None,
                    &item.blob_sha256,
                    &item.file_name,
                    &item.extension,
                    &item.mime_type,
                    timestamp,
                )?;

                super::attachments::add_attachment_reference_in_transaction(
                    &transaction,
                    chat_id,
                    &version.id,
                    None,
                    "composer",
                    position,
                    timestamp,
                )?;

                position = position
                    .checked_add(1)
                    .ok_or_else(|| "Attachment position overflow".to_string())?;
            }

            transaction
                .execute(
                    "UPDATE storage_operations
                     SET state = 'COMPLETED', updated_at = ?1, error_message = NULL
                     WHERE id = ?2",
                    params![now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot complete import journal: {error}"))?;

            Ok(())
        })();

        if let Err(error) = commit_result {
            drop(transaction);

            let _ = self.connection.execute(
                "UPDATE storage_operations
                 SET state = 'FAILED', error_message = ?1, updated_at = ?2
                 WHERE id = ?3",
                params![error, now_timestamp().unwrap_or(timestamp), operation_id],
            );

            return Err(error);
        }

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit imported attachments: {error}"))?;

        self.list_composer_attachments(chat_id)
    }
}

fn process_selected_paths(app: AppHandle, paths: Vec<PathBuf>, chat_id: String) {
    tauri::async_runtime::spawn_blocking(move || {
        let result = (|| -> Result<Vec<AttachmentReferenceRecord>, String> {
            let state = app.state::<Mutex<Storage>>();
            let mut storage = state
                .lock()
                .map_err(|error| format!("Storage lock poisoned: {error}"))?;

            storage.import_selected_paths(&chat_id, paths)
        })();

        match result {
            Ok(attachments) => {
                let _ = app.emit("attachments-imported", attachments);
            }
            Err(error) => {
                // Keep the frontend draft synchronized even after a failed batch.
                let current = (|| {
                    let state = app.state::<Mutex<Storage>>();
                    let storage = state.lock().ok()?;
                    storage.list_composer_attachments(&chat_id).ok()
                })();

                if let Some(attachments) = current {
                    let _ = app.emit("attachments-imported", attachments);
                }

                let _ = app.emit("attachments-import-error", error);
            }
        }
    });
}

#[tauri::command]
pub fn import_attachments(app: AppHandle, chat_id: String) -> Result<(), String> {
    let callback_app = app.clone();

    app.dialog().file().pick_files(move |selected_files| {
        if let Some(selected_files) = selected_files {
            let paths = selected_files
                .into_iter()
                .map(|file| file.into_path())
                .collect::<Result<Vec<_>, _>>();

            match paths {
                Ok(paths) => process_selected_paths(callback_app, paths, chat_id),
                Err(error) => {
                    let _ = callback_app.emit(
                        "attachments-import-error",
                        format!("Cannot resolve selected files: {error}"),
                    );
                }
            }
        }
    });

    Ok(())
}

    if paths.is_empty() {
        return Ok(());
    }

    process_selected_paths(app, paths.into_iter().map(PathBuf::from).collect(), chat_id);

    Ok(())
}

use tauri::{

    http::{Method, Request, Response, StatusCode, header},

use super::Storage;

const MAX_RESPONSE_SIZE: u64 = 2 * 1024 * 1024;

pub fn handle_preview_request<R: Runtime>(
    app: AppHandle<R>,
    request: Request<Vec<u8>>,
    responder: UriSchemeResponder,
) {
    let response = match build_preview_response(&app, &request) {
        Ok(response) => response,
        Err(status) => empty_response(status),
    };

    let _ = responder.respond(response);
}

fn build_preview_response<R: Runtime>(
    app: &AppHandle<R>,
    request: &Request<Vec<u8>>,
) -> Result<Response<Vec<u8>>, StatusCode> {
    if request.method() != Method::GET && request.method() != Method::HEAD {
        return Err(StatusCode::METHOD_NOT_ALLOWED);
    }

    let segments: Vec<&str> = request.uri().path().trim_matches('/').split('/').collect();

    if segments.len() != 2
        || segments.iter().any(|segment| {
            segment.is_empty()
                || !segment
                    .bytes()
                    .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
        })
    {
        return Err(StatusCode::BAD_REQUEST);
    }

    let chat_id = segments[0];
    let reference_id = segments[1];

    let state: State<'_, Mutex<Storage>> = app.state();
    let (path, file_name, mime_type, file_size) = {
        let storage = state
            .lock()
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

        let preview = storage
            .get_preview_blob(chat_id, reference_id)
            .map_err(|_| StatusCode::NOT_FOUND)?;

        let path = storage
            .resolve_blob_path(&preview.relative_path)
            .map_err(|_| StatusCode::NOT_FOUND)?;

        (
            path,
            preview.file_name,
            safe_mime_type(&preview.mime_type).to_string(),
            preview.original_size as u64,
        )
    };

    let range_header = request.headers().get(header::RANGE);
    let (start, end, status) = match range_header {
        Some(value) => {
            let value = value
                .to_str()
                .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;
            let (start, end) = parse_range(value, file_size)?;
            (start, end, StatusCode::PARTIAL_CONTENT)
        }
        None => {
            if file_size > MAX_RESPONSE_SIZE {
                return Err(StatusCode::RANGE_NOT_SATISFIABLE);
            }
            (0, file_size.saturating_sub(1), StatusCode::OK)
        }
    };

    let length = if file_size == 0 { 0 } else { end - start + 1 };

    if length > MAX_RESPONSE_SIZE {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let mut body = Vec::new();

    if request.method() == Method::GET && length > 0 {
        let mut file = File::open(path).map_err(|_| StatusCode::NOT_FOUND)?;
        file.seek(SeekFrom::Start(start))
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

        body.resize(length as usize, 0);
        file.read_exact(&mut body)
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
    }

    let disposition = format!("inline; filename=\"{}\"", safe_filename(&file_name));

    let mut builder = Response::builder()
        .status(status)
        .header(header::CONTENT_TYPE, mime_type)
        .header(header::CONTENT_LENGTH, length.to_string())
        .header(header::ACCEPT_RANGES, "bytes")
        .header(header::CONTENT_DISPOSITION, disposition)
        .header("X-Content-Type-Options", "nosniff")
        .header("Content-Security-Policy", "default-src 'none'; sandbox");

    if status == StatusCode::PARTIAL_CONTENT {
        builder = builder.header(
            header::CONTENT_RANGE,
            format!("bytes {start}-{end}/{file_size}"),
        );
    }

    builder
        .body(body)
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)
}

fn parse_range(value: &str, size: u64) -> Result<(u64, u64), StatusCode> {
    if size == 0 {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let value = value
        .strip_prefix("bytes=")
        .ok_or(StatusCode::RANGE_NOT_SATISFIABLE)?;

    if value.contains(',') {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let (start, end) = value
        .split_once('-')
        .ok_or(StatusCode::RANGE_NOT_SATISFIABLE)?;

    let (start, end) = if start.is_empty() {
        let suffix: u64 = end.parse().map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;

        if suffix == 0 {
            return Err(StatusCode::RANGE_NOT_SATISFIABLE);
        }

        (size.saturating_sub(suffix), size - 1)
    } else {
        let start: u64 = start
            .parse()
            .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;

        let end = if end.is_empty() {
            size - 1
        } else {
            end.parse::<u64>()
                .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?
                .min(size - 1)
        };

        (start, end)
    };

    if start >= size || end < start || end - start + 1 > MAX_RESPONSE_SIZE {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    Ok((start, end))
}

fn safe_mime_type(mime: &str) -> &str {
    match mime {
        "application/pdf" | "image/png" | "image/jpeg" | "image/webp" | "image/gif"
        | "image/bmp" | "text/plain" | "text/markdown" => mime,
        _ => "application/octet-stream",
    }
}

fn safe_filename(name: &str) -> String {
    name.chars()
        .filter(|character| {
            character.is_ascii_alphanumeric() || matches!(character, '.' | '-' | '_' | ' ')
        })
        .take(120)
        .collect::<String>()
        .replace('"', "")
}

fn empty_response(status: StatusCode) -> Response<Vec<u8>> {
    Response::builder()
        .status(status)
        .header(header::CONTENT_LENGTH, "0")
        .header("X-Content-Type-Options", "nosniff")
        .body(Vec::new())
        .unwrap_or_else(|_| Response::new(Vec::new()))
}

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))

pub(super) fn create_attachment_version_in_transaction(
    tx: &Transaction<'_>,
    attachment_id: Option<&str>,
    blob_sha256: &str,
    file_name: &str,
    extension: &str,
    mime_type: &str,
    timestamp: i64,
) -> Result<AttachmentVersionRecord, String> {
    if file_name.trim().is_empty() {
        return Err("Attachment filename cannot be empty".to_string());
    }

    let original_size: i64 = tx
        .query_row(
            "SELECT original_size FROM blobs WHERE sha256 = ?1",
            params![blob_sha256],
            |row| row.get(0),
        )
        .optional()
        .map_err(|e| format!("Cannot verify attachment blob: {e}"))?
        .ok_or_else(|| "Attachment blob does not exist".to_string())?;

    let resolved_id = match attachment_id {
        Some(id) => {
            let exists: bool = tx
                .query_row(
                    "SELECT EXISTS(SELECT 1 FROM attachments WHERE id = ?1)",
                    params![id],
                    |row| row.get(0),
                )
                .map_err(|e| format!("Cannot verify attachment: {e}"))?;
            if !exists {
                return Err(format!("Attachment not found: {id}"));
            }
            id.to_owned()
        }
        None => {
            let id = Uuid::new_v4().to_string();
            tx.execute(
                "INSERT INTO attachments (id, created_at, updated_at) VALUES (?1, ?2, ?2)",
                params![id, timestamp],
            )
            .map_err(|e| format!("Cannot create attachment: {e}"))?;
            id
        }
    };

    let version_number: i64 = tx
        .query_row(
            "SELECT COALESCE(MAX(version_number), 0) + 1
         FROM attachment_versions WHERE attachment_id = ?1",
            params![resolved_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("Cannot determine attachment version: {e}"))?;

    let version_id = Uuid::new_v4().to_string();

    tx.execute(
        "INSERT INTO attachment_versions (
            id, attachment_id, version_number, blob_sha256,
            file_name, extension, mime_type, created_at
        ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
        params![
            version_id,
            resolved_id,
            version_number,
            blob_sha256,
            file_name,
            extension,
            mime_type,
            timestamp
        ],
    )
    .map_err(|e| format!("Cannot save attachment version: {e}"))?;

    tx.execute(
        "UPDATE attachments SET updated_at = ?1 WHERE id = ?2",
        params![timestamp, resolved_id],
    )
    .map_err(|e| format!("Cannot update attachment timestamp: {e}"))?;

    Ok(AttachmentVersionRecord {
        id: version_id,
        attachment_id: resolved_id,
        version_number,
        blob_sha256: blob_sha256.to_owned(),
        file_name: file_name.to_owned(),
        extension: extension.to_owned(),
        mime_type: mime_type.to_owned(),
        original_size,
        created_at: timestamp,
    })
}

pub(super) fn add_attachment_reference_in_transaction(
    tx: &Transaction<'_>,
    chat_id: &str,
    version_id: &str,
    message_id: Option<&str>,
    purpose: &str,
    position: i64,
    timestamp: i64,
) -> Result<String, String> {
    if !matches!(purpose, "composer" | "message") {
        return Err("Invalid attachment reference purpose".to_string());
    }
    if position < 0 {
        return Err("Attachment position cannot be negative".to_string());
    }
    match (purpose, message_id) {
        ("composer", None) | ("message", Some(_)) => {}
        _ => {
            return Err(
                "Composer references cannot have a message ID, and message references require one"
                    .to_string(),
            );
        }
    }

    let chat_exists: bool = tx
        .query_row(
            "SELECT EXISTS(SELECT 1 FROM chats WHERE id = ?1)",
            params![chat_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("Cannot verify reference chat: {e}"))?;
    if !chat_exists {
        return Err(format!("Chat not found: {chat_id}"));
    }

    let version_exists: bool = tx
        .query_row(
            "SELECT EXISTS(SELECT 1 FROM attachment_versions WHERE id = ?1)",
            params![version_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("Cannot verify attachment version: {e}"))?;
    if !version_exists {
        return Err(format!("Attachment version not found: {version_id}"));
    }

    if let Some(message_id) = message_id {
        let belongs: bool = tx
            .query_row(
                "SELECT EXISTS(
                SELECT 1 FROM messages WHERE id = ?1 AND chat_id = ?2
            )",
                params![message_id, chat_id],
                |row| row.get(0),
            )
            .map_err(|e| format!("Cannot verify target message: {e}"))?;
        if !belongs {
            return Err("Target message does not belong to the specified chat".to_string());
        }
    }

    let reference_id = Uuid::new_v4().to_string();
    tx.execute(
        "INSERT INTO attachment_references (
            id, chat_id, attachment_version_id, message_id,
            purpose, position, created_at
        ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7)",
        params![
            reference_id,
            chat_id,
            version_id,
            message_id,
            purpose,
            position,
            timestamp
        ],
    )
    .map_err(|e| format!("Cannot create attachment reference: {e}"))?;

    tx.execute(
        "UPDATE chats SET updated_at = ?1 WHERE id = ?2",
        params![timestamp, chat_id],
    )
    .map_err(|e| format!("Cannot update chat timestamp: {e}"))?;

    Ok(reference_id)
}

    /// Creates a new attachment or adds a version to an existing attachment.
    /// The referenced physical blob must already exist.
    pub fn create_attachment_version(
        &mut self,

        blob_sha256: &str,
        file_name: &str,
        extension: &str,
        mime_type: &str,

        let timestamp = now_timestamp()?;

            file_name,
            extension,
            mime_type,
            timestamp,

            .map_err(|e| format!("Cannot commit attachment version: {e}"))?;

    }

impl Storage {
    pub fn promote_composer_attachments(
        &mut self,
        chat_id: &str,
        message_id: &str,
    ) -> Result<Vec<AttachmentReferenceRecord>, String> {
        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start attachment promotion: {error}"))?;

        let message_exists: bool = transaction
            .query_row(
                "SELECT EXISTS(
                    SELECT 1 FROM messages
                    WHERE id = ?1 AND chat_id = ?2
                )",
                rusqlite::params![message_id, chat_id],
                |row| row.get(0),
            )
            .map_err(|error| format!("Cannot verify target message: {error}"))?;

        if !message_exists {
            return Err("Target message does not belong to the specified chat".into());
        }

        transaction
            .execute(
                "UPDATE attachment_references
                 SET message_id = ?1, purpose = 'message'
                 WHERE chat_id = ?2 AND purpose = 'composer'",
                rusqlite::params![message_id, chat_id],
            )
            .map_err(|error| format!("Cannot promote composer attachments: {error}"))?;

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit attachment promotion: {error}"))?;

        self.list_message_attachments(message_id)
    }
}

#[tauri::command]
pub fn read_attachment_chunk(
    state: tauri::State<'_, std::sync::Mutex<Storage>>,
    chat_id: String,
    reference_id: String,
    offset: u64,
    length: usize,
) -> Result<Vec<u8>, String> {
    const MAX_CHUNK_SIZE: usize = 256 * 1024;

    if length == 0 || length > MAX_CHUNK_SIZE {
        return Err(format!(
            "Chunk length must be between 1 and {} bytes",
            MAX_CHUNK_SIZE
        ));
    }

    let storage = state
        .lock()
        .map_err(|error| format!("Storage lock poisoned: {error}"))?;

    // The reference is resolved within the requested chat.
    let blob = storage.get_preview_blob(&chat_id, &reference_id)?;
    let path = storage.resolve_blob_path(&blob.relative_path)?;

    use std::io::{Read, Seek, SeekFrom};

    let mut file =
        std::fs::File::open(&path).map_err(|error| format!("Cannot open attachment: {error}"))?;

    let file_size = file
        .metadata()
        .map_err(|error| format!("Cannot inspect attachment: {error}"))?
        .len();

    if offset > file_size {
        return Err("Attachment offset is out of bounds".to_owned());
    }

    let bytes_to_read = (file_size - offset).min(length as u64) as usize;

    file.seek(SeekFrom::Start(offset))
        .map_err(|error| format!("Cannot seek attachment: {error}"))?;

    let mut buffer = vec![0u8; bytes_to_read];
    file.read_exact(&mut buffer)
        .map_err(|error| format!("Cannot read attachment chunk: {error}"))?;

    Ok(buffer)
}

    tx: &Transaction<'_>,
    chat_id: &str,
    version_id: &str,
    message_id: Option<&str>,
    purpose: &str,
    position: i64,
    timestamp: i64,
) -> Result<String, String> {
    if !matches!(purpose, "composer" | "message") {
        return Err("Invalid attachment reference purpose".to_string());
    }
    if position < 0 {
        return Err("Attachment position cannot be negative".to_string());
    }
    match (purpose, message_id) {
        ("composer", None) | ("message", Some(_)) => {}
        _ => {
            return Err(
                "Composer references cannot have a message ID, and message references require one"
                    .to_string(),
            );
        }
    }

    let chat_exists: bool = tx
        .query_row(
            "SELECT EXISTS(SELECT 1 FROM chats WHERE id = ?1)",
            params![chat_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("Cannot verify reference chat: {e}"))?;
    if !chat_exists {
        return Err(format!("Chat not found: {chat_id}"));
    }

    let version_exists: bool = tx
        .query_row(
            "SELECT EXISTS(SELECT 1 FROM attachment_versions WHERE id = ?1)",
            params![version_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("Cannot verify attachment version: {e}"))?;
    if !version_exists {
        return Err(format!("Attachment version not found: {version_id}"));
    }

    if let Some(message_id) = message_id {
        let belongs: bool = tx
            .query_row(
                "SELECT EXISTS(
                SELECT 1 FROM messages WHERE id = ?1 AND chat_id = ?2
            )",
                params![message_id, chat_id],
                |row| row.get(0),
            )
            .map_err(|e| format!("Cannot verify target message: {e}"))?;
        if !belongs {
            return Err("Target message does not belong to the specified chat".to_string());
        }
    }

    let reference_id = Uuid::new_v4().to_string();
    tx.execute(
        "INSERT INTO attachment_references (
            id, chat_id, attachment_version_id, message_id,
            purpose, position, created_at
        ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7)",
        params![
            reference_id,
            chat_id,
            version_id,
            message_id,
            purpose,
            position,
            timestamp
        ],
    )
    .map_err(|e| format!("Cannot create attachment reference: {e}"))?;

    tx.execute(
        "UPDATE chats SET updated_at = ?1 WHERE id = ?2",
        params![timestamp, chat_id],
    )
    .map_err(|e| format!("Cannot update chat timestamp: {e}"))?;

    Ok(reference_id)
}

#[tauri::command]
pub fn read_attachment_chunk(
    state: tauri::State<'_, std::sync::Mutex<Storage>>,
    chat_id: String,
    reference_id: String,
    offset: u64,
    length: usize,
) -> Result<Vec<u8>, String> {
    const MAX_CHUNK_SIZE: usize = 256 * 1024;

    if length == 0 || length > MAX_CHUNK_SIZE {
        return Err(format!(
            "Chunk length must be between 1 and {} bytes",
            MAX_CHUNK_SIZE
        ));
    }

    let storage = state
        .lock()
        .map_err(|error| format!("Storage lock poisoned: {error}"))?;

    // The reference is resolved within the requested chat.
    let blob = storage.get_preview_blob(&chat_id, &reference_id)?;
    let path = storage.resolve_blob_path(&blob.relative_path)?;

    use std::io::{Read, Seek, SeekFrom};

    let mut file =
        std::fs::File::open(&path).map_err(|error| format!("Cannot open attachment: {error}"))?;

    let file_size = file
        .metadata()
        .map_err(|error| format!("Cannot inspect attachment: {error}"))?
        .len();

    if offset > file_size {
        return Err("Attachment offset is out of bounds".to_owned());
    }

    let bytes_to_read = (file_size - offset).min(length as u64) as usize;

    file.seek(SeekFrom::Start(offset))
        .map_err(|error| format!("Cannot seek attachment: {error}"))?;

    let mut buffer = vec![0u8; bytes_to_read];
    file.read_exact(&mut buffer)
        .map_err(|error| format!("Cannot read attachment chunk: {error}"))?;

    Ok(buffer)
}

use std::{
    fs::{self, File, OpenOptions},
    io::{Read, Write},
    path::{Component, Path, PathBuf},
    time::{SystemTime, UNIX_EPOCH},
};

use rusqlite::{OptionalExtension, params};
use serde::Serialize;
use sha2::{Digest, Sha256};
use uuid::Uuid;

use super::Storage;

const BUFFER_SIZE: usize = 1024 * 1024;

#[derive(Debug, Clone, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct BlobRecord {
    pub sha256: String,
    pub relative_path: String,
    pub original_size: i64,
    pub stored_size: i64,
}

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))
}

fn digest_to_hex(digest: &[u8]) -> String {
    digest.iter().map(|byte| format!("{byte:02x}")).collect()
}

fn hash_file(path: &Path) -> Result<String, String> {
    let mut file =
        File::open(path).map_err(|error| format!("Cannot open existing blob: {error}"))?;

    let mut hasher = Sha256::new();
    let mut buffer = vec![0u8; BUFFER_SIZE];

    loop {
        let bytes_read = file
            .read(&mut buffer)
            .map_err(|error| format!("Cannot verify existing blob: {error}"))?;

        if bytes_read == 0 {
            break;
        }

        hasher.update(&buffer[..bytes_read]);
    }

    Ok(digest_to_hex(&hasher.finalize()))
}

fn resolve_relative_path(root: &Path, relative: &str) -> Result<PathBuf, String> {
    let path = Path::new(relative);

    if path.is_absolute()
        || !path
            .components()
            .all(|component| matches!(component, Component::Normal(_)))
        || path.components().next() != Some(Component::Normal("blobs".as_ref()))
    {
        return Err("Invalid internal blob path".to_string());
    }

    Ok(root.join(path))
}

impl Storage {
    pub fn resolve_blob_path(&self, relative: &str) -> Result<PathBuf, String> {
        let candidate = resolve_relative_path(&self.root_dir, relative)?;

        let storage_root = self
            .root_dir
            .canonicalize()
            .map_err(|error| format!("Cannot resolve storage directory: {error}"))?;

        let blobs_root = self
            .root_dir
            .join("blobs")
            .canonicalize()
            .map_err(|error| format!("Cannot resolve blob storage directory: {error}"))?;

        if !blobs_root.starts_with(&storage_root) {
            return Err("Blob storage directory escapes storage root".to_string());
        }

        let canonical = candidate
            .canonicalize()
            .map_err(|error| format!("Cannot resolve blob file: {error}"))?;

        if !canonical.starts_with(&blobs_root) {
            return Err("Blob path escapes storage directory".to_string());
        }

        let metadata = fs::symlink_metadata(&candidate)
            .map_err(|error| format!("Cannot inspect blob file: {error}"))?;

        if metadata.file_type().is_symlink() || !metadata.is_file() {
            return Err("Blob path is not a regular file".to_string());
        }

        Ok(canonical)
    }

    pub fn store_blob<R: Read>(&mut self, mut reader: R) -> Result<BlobRecord, String> {
        let operation_id = Uuid::new_v4().to_string();
        let timestamp = now_timestamp()?;

        let temp_relative = format!("temp/{operation_id}.part");
        let temp_path = self.root_dir.join(&temp_relative);

        self.connection
            .execute(
                "INSERT INTO storage_operations (
                    id, operation_type, state, created_at, updated_at
                 ) VALUES (?1, 'store_blob', 'PENDING', ?2, ?2)",
                params![operation_id, timestamp],
            )
            .map_err(|error| format!("Cannot create storage journal entry: {error}"))?;

        self.connection
            .execute(
                "UPDATE storage_operations
                 SET state = 'PROCESSING', source_path = ?1, updated_at = ?2
                 WHERE id = ?3",
                params![temp_relative, timestamp, operation_id],
            )
            .map_err(|error| format!("Cannot update storage journal: {error}"))?;

        let mut output = OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(&temp_path)
            .map_err(|error| format!("Cannot create temporary blob: {error}"))?;

        let mut hasher = Sha256::new();
        let mut buffer = vec![0u8; BUFFER_SIZE];
        let mut total_size: u64 = 0;

        loop {
            let bytes_read = reader
                .read(&mut buffer)
                .map_err(|error| format!("Cannot read incoming file: {error}"))?;

            if bytes_read == 0 {
                break;
            }

            output
                .write_all(&buffer[..bytes_read])
                .map_err(|error| format!("Cannot write temporary blob: {error}"))?;

            hasher.update(&buffer[..bytes_read]);

            total_size = total_size
                .checked_add(bytes_read as u64)
                .ok_or_else(|| "File size overflow".to_string())?;
        }

        output
            .sync_all()
            .map_err(|error| format!("Cannot synchronize temporary blob: {error}"))?;

        drop(output);

        let original_size = i64::try_from(total_size)
            .map_err(|error| format!("File is too large for SQLite metadata: {error}"))?;

        let sha256 = digest_to_hex(&hasher.finalize());

        let existing: Option<(String, i64)> = self
            .connection
            .query_row(
                "SELECT relative_path, original_size
                 FROM blobs WHERE sha256 = ?1",
                params![sha256],
                |row| Ok((row.get(0)?, row.get(1)?)),
            )
            .optional()
            .map_err(|error| format!("Cannot check blob deduplication: {error}"))?;

        let relative_path = if let Some((existing_relative, _)) = existing {
            let existing_path = resolve_relative_path(&self.root_dir, &existing_relative)?;

            let existing_is_valid = existing_path.is_file()
                && hash_file(&existing_path)
                    .map(|hash| hash == sha256)
                    .unwrap_or(false);

            if existing_is_valid {
                fs::remove_file(&temp_path)
                    .map_err(|error| format!("Cannot remove duplicate temporary file: {error}"))?;

                self.connection
                    .execute(
                        "UPDATE blobs SET last_accessed_at = ?1 WHERE sha256 = ?2",
                        params![now_timestamp()?, sha256],
                    )
                    .map_err(|error| format!("Cannot update blob access time: {error}"))?;

                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET blob_sha256 = ?1, state = 'COMPLETED', updated_at = ?2
                         WHERE id = ?3",
                        params![sha256, now_timestamp()?, operation_id],
                    )
                    .map_err(|error| format!("Cannot complete deduplication journal: {error}"))?;

                return Ok(BlobRecord {
                    sha256,
                    relative_path: existing_relative,
                    original_size,
                    stored_size: original_size,
                });
            }

            if let Some(parent) = existing_path.parent() {
                fs::create_dir_all(parent)
                    .map_err(|error| format!("Cannot recreate blob directory: {error}"))?;
            }

            self.connection
                .execute(
                    "UPDATE storage_operations
                     SET state = 'COMMITTING', target_path = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![existing_relative, now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot update repair journal: {error}"))?;

            fs::rename(&temp_path, &existing_path)
                .map_err(|error| format!("Cannot repair existing blob: {error}"))?;

            self.connection
                .execute(
                    "UPDATE blobs
                     SET original_size = ?1, stored_size = ?1,
                         storage_state = 'original',
                         compression_format = NULL,
                         last_accessed_at = ?2,
                         verified_at = ?2
                     WHERE sha256 = ?3",
                    params![original_size, now_timestamp()?, sha256],
                )
                .map_err(|error| format!("Cannot update repaired blob metadata: {error}"))?;

            existing_relative
        } else {
            let prefix = &sha256[..2];
            let relative = format!("blobs/{prefix}/{sha256}.blob");
            let target_path = resolve_relative_path(&self.root_dir, &relative)?;

            if let Some(parent) = target_path.parent() {
                fs::create_dir_all(parent)
                    .map_err(|error| format!("Cannot create blob directory: {error}"))?;
            }

            self.connection
                .execute(
                    "UPDATE storage_operations
                     SET state = 'VERIFYING', target_path = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![relative, now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot update verification journal: {error}"))?;

            let orphan_is_valid = target_path.is_file()
                && hash_file(&target_path)
                    .map(|hash| hash == sha256)
                    .unwrap_or(false);

            if orphan_is_valid {
                fs::remove_file(&temp_path)
                    .map_err(|error| format!("Cannot remove duplicate orphan blob: {error}"))?;
            } else {
                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET state = 'COMMITTING', updated_at = ?1
                         WHERE id = ?2",
                        params![now_timestamp()?, operation_id],
                    )
                    .map_err(|error| format!("Cannot update commit journal: {error}"))?;

                fs::rename(&temp_path, &target_path)
                    .map_err(|error| format!("Cannot commit blob file: {error}"))?;
            }

            relative
        };

        let timestamp = now_timestamp()?;

        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start blob metadata transaction: {error}"))?;

        transaction
            .execute(
                "INSERT INTO blobs (
                    sha256, relative_path, original_size, stored_size,
                    storage_state, created_at, last_accessed_at, verified_at
                 ) VALUES (?1, ?2, ?3, ?3, 'original', ?4, ?4, ?4)
                 ON CONFLICT(sha256) DO UPDATE SET
                    relative_path = excluded.relative_path,
                    original_size = excluded.original_size,
                    stored_size = excluded.stored_size,
                    storage_state = 'original',
                    compression_format = NULL,
                    last_accessed_at = excluded.last_accessed_at,
                    verified_at = excluded.verified_at",
                params![sha256, relative_path, original_size, timestamp],
            )
            .map_err(|error| format!("Cannot save blob metadata: {error}"))?;

        transaction
            .execute(
                "UPDATE storage_operations
                 SET blob_sha256 = ?1, state = 'COMPLETED', updated_at = ?2
                 WHERE id = ?3",
                params![sha256, timestamp, operation_id],
            )
            .map_err(|error| format!("Cannot complete blob journal: {error}"))?;

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit blob metadata: {error}"))?;

        Ok(BlobRecord {
            sha256,
            relative_path,
            original_size,
            stored_size: original_size,
        })
    }

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))
}

fn file_metadata(path: &Path) -> Result<(String, String, String), String> {
    let file_name = path
        .file_name()
        .ok_or_else(|| "Selected file has no filename".to_string())?
        .to_string_lossy()
        .into_owned();

    let extension = path
        .extension()
        .and_then(|value| value.to_str())
        .unwrap_or("")
        .to_ascii_lowercase();

    let mime_type = match extension.as_str() {
        "pdf" => "application/pdf",
        "png" => "image/png",
        "jpg" | "jpeg" => "image/jpeg",
        "webp" => "image/webp",
        "gif" => "image/gif",
        "bmp" => "image/bmp",
        "tif" | "tiff" => "image/tiff",
        "avif" => "image/avif",
        "heic" | "heif" => "image/heic",
        "txt" | "log" | "csv" | "tsv" => "text/plain",
        "md" | "markdown" => "text/markdown",
        "html" | "htm" => "text/html",
        "json" => "application/json",
        "xml" => "application/xml",
        "js" | "mjs" | "cjs" => "text/javascript",
        "ts" | "tsx" => "text/typescript",
        "css" => "text/css",
        "rs" => "text/rust",
        "py" => "text/x-python",
        "java" => "text/x-java-source",
        "c" | "h" => "text/x-c",
        "cpp" | "hpp" | "cc" => "text/x-c++",
        "sh" | "bash" => "text/x-shellscript",
        "yaml" | "yml" => "application/yaml",
        "toml" => "application/toml",
        "docx" => "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
        "pptx" => "application/vnd.openxmlformats-officedocument.presentationml.presentation",
        "xlsx" => "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        "odt" => "application/vnd.oasis.opendocument.text",
        "ods" => "application/vnd.oasis.opendocument.spreadsheet",
        "odp" => "application/vnd.oasis.opendocument.presentation",
        "epub" => "application/epub+zip",
        "zip" => "application/zip",
        "gz" => "application/gzip",
        "tar" => "application/x-tar",
        "rtf" => "application/rtf",
        _ => "application/octet-stream",
    };

    Ok((file_name, extension, mime_type.to_string()))
}

struct PreparedAttachment {
    blob_sha256: String,
    file_name: String,
    extension: String,
    mime_type: String,
}

impl Storage {
    fn import_selected_paths(
        &mut self,
        chat_id: &str,
        paths: Vec<PathBuf>,
    ) -> Result<Vec<AttachmentReferenceRecord>, String> {
        if paths.is_empty() {
            return self.list_composer_attachments(chat_id);
        }

        let timestamp = now_timestamp()?;
        let operation_id = Uuid::new_v4().to_string();

        let chat_exists: bool = self
            .connection
            .query_row(
                "SELECT EXISTS(SELECT 1 FROM chats WHERE id = ?1)",
                params![chat_id],
                |row| row.get(0),
            )
            .map_err(|error| format!("Cannot verify import chat: {error}"))?;

        if !chat_exists {
            return Err(format!("Chat not found: {chat_id}"));
        }

        let existing = self.list_composer_attachments(chat_id)?;
        let mut position = i64::try_from(existing.len())
            .map_err(|error| format!("Attachment position overflow: {error}"))?;

        self.connection
            .execute(
                "INSERT INTO storage_operations (
                    id, operation_type, state, metadata_json,
                    created_at, updated_at
                ) VALUES (?1, 'import_batch', 'PENDING', ?2, ?3, ?3)",
                params![
                    operation_id,
                    json!({
                        "chatId": chat_id,
                        "files": []
                    })
                    .to_string(),
                    timestamp
                ],
            )
            .map_err(|error| format!("Cannot create import journal: {error}"))?;

        self.connection
            .execute(
                "UPDATE storage_operations
                 SET state = 'PROCESSING', updated_at = ?1
                 WHERE id = ?2",
                params![now_timestamp()?, operation_id],
            )
            .map_err(|error| format!("Cannot start import journal: {error}"))?;

        let preparation = (|| -> Result<Vec<PreparedAttachment>, String> {
            let mut prepared = Vec::with_capacity(paths.len());

            for path in paths {
                let metadata = std::fs::metadata(&path)
                    .map_err(|error| format!("Cannot inspect file {}: {error}", path.display()))?;

                if !metadata.is_file() {
                    return Err(format!("Not a regular file: {}", path.display()));
                }

                let (file_name, extension, mime_type) = file_metadata(&path)?;

                let file = File::open(&path)
                    .map_err(|error| format!("Cannot open {file_name}: {error}"))?;

                let blob = self.store_blob(file)?;

                prepared.push(PreparedAttachment {
                    blob_sha256: blob.sha256,
                    file_name,
                    extension,
                    mime_type,
                });

                let files_json: Vec<_> = prepared
                    .iter()
                    .map(|item| {
                        json!({
                            "blobSha256": item.blob_sha256,
                            "fileName": item.file_name,
                            "extension": item.extension,
                            "mimeType": item.mime_type
                        })
                    })
                    .collect();

                self.connection
                    .execute(
                        "UPDATE storage_operations
                         SET metadata_json = ?1, updated_at = ?2
                         WHERE id = ?3",
                        params![
                            json!({
                                "chatId": chat_id,
                                "files": files_json
                            })
                            .to_string(),
                            now_timestamp()?,
                            operation_id
                        ],
                    )
                    .map_err(|error| format!("Cannot update import journal: {error}"))?;
            }

            Ok(prepared)
        })();

        let prepared = match preparation {
            Ok(prepared) => prepared,
            Err(error) => {
                let _ = self.connection.execute(
                    "UPDATE storage_operations
                     SET state = 'FAILED', error_message = ?1, updated_at = ?2
                     WHERE id = ?3",
                    params![error, now_timestamp().unwrap_or(timestamp), operation_id],
                );
                return Err(error);
            }
        };

        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start atomic attachment transaction: {error}"))?;

        let commit_result = (|| -> Result<(), String> {
            for item in &prepared {
                let version = super::attachments::create_attachment_version_in_transaction(
                    &transaction,
                    None,
                    &item.blob_sha256,
                    &item.file_name,
                    &item.extension,
                    &item.mime_type,
                    timestamp,
                )?;

                super::attachments::add_attachment_reference_in_transaction(
                    &transaction,
                    chat_id,
                    &version.id,
                    None,
                    "composer",
                    position,
                    timestamp,
                )?;

                position = position
                    .checked_add(1)
                    .ok_or_else(|| "Attachment position overflow".to_string())?;
            }

            transaction
                .execute(
                    "UPDATE storage_operations
                     SET state = 'COMPLETED', updated_at = ?1, error_message = NULL
                     WHERE id = ?2",
                    params![now_timestamp()?, operation_id],
                )
                .map_err(|error| format!("Cannot complete import journal: {error}"))?;

            Ok(())
        })();

        if let Err(error) = commit_result {
            drop(transaction);

            let _ = self.connection.execute(
                "UPDATE storage_operations
                 SET state = 'FAILED', error_message = ?1, updated_at = ?2
                 WHERE id = ?3",
                params![error, now_timestamp().unwrap_or(timestamp), operation_id],
            );

            return Err(error);
        }

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit imported attachments: {error}"))?;

        self.list_composer_attachments(chat_id)
    }
}

fn process_selected_paths(app: AppHandle, paths: Vec<PathBuf>, chat_id: String) {
    tauri::async_runtime::spawn_blocking(move || {
        let result = (|| -> Result<Vec<AttachmentReferenceRecord>, String> {
            let state = app.state::<Mutex<Storage>>();
            let mut storage = state
                .lock()
                .map_err(|error| format!("Storage lock poisoned: {error}"))?;

            storage.import_selected_paths(&chat_id, paths)
        })();

        match result {
            Ok(attachments) => {
                let _ = app.emit("attachments-imported", attachments);
            }
            Err(error) => {
                // Keep the frontend draft synchronized even after a failed batch.
                let current = (|| {
                    let state = app.state::<Mutex<Storage>>();
                    let storage = state.lock().ok()?;
                    storage.list_composer_attachments(&chat_id).ok()
                })();

                if let Some(attachments) = current {
                    let _ = app.emit("attachments-imported", attachments);
                }

                let _ = app.emit("attachments-import-error", error);
            }
        }
    });
}

#[tauri::command]
pub fn import_attachments(app: AppHandle, chat_id: String) -> Result<(), String> {
    let callback_app = app.clone();

    app.dialog().file().pick_files(move |selected_files| {
        if let Some(selected_files) = selected_files {
            let paths = selected_files
                .into_iter()
                .map(|file| file.into_path())
                .collect::<Result<Vec<_>, _>>();

            match paths {
                Ok(paths) => process_selected_paths(callback_app, paths, chat_id),
                Err(error) => {
                    let _ = callback_app.emit(
                        "attachments-import-error",
                        format!("Cannot resolve selected files: {error}"),
                    );
                }
            }
        }
    });

    Ok(())
}

#[tauri::command]
pub fn import_files(app: AppHandle, chat_id: String, paths: Vec<String>) -> Result<(), String> {
    if paths.is_empty() {
        return Ok(());
    }

    process_selected_paths(app, paths.into_iter().map(PathBuf::from).collect(), chat_id);

    Ok(())
}

use std::{
    fs,
    path::{Path, PathBuf},
    time::Duration,
};

use rusqlite::{Connection, OpenFlags};
use schema::{INITIAL_SCHEMA, SCHEMA_VERSION};

pub struct Storage {
    connection: Connection,
    root_dir: PathBuf,
}

impl Storage {
    pub fn initialize(app_data_dir: &Path) -> Result<Self, String> {
        let root_dir = app_data_dir.join("storage");

        for directory in [
            root_dir.join("blobs"),
            root_dir.join("temp"),
            root_dir.join("extracted"),
            root_dir.join("indexes"),
        ] {
            fs::create_dir_all(&directory)
                .map_err(|error| format!("Cannot create storage directory: {error}"))?;
        }

        let database_path = root_dir.join("sonne.db");

        let connection = Connection::open_with_flags(
            &database_path,
            OpenFlags::SQLITE_OPEN_READ_WRITE
                | OpenFlags::SQLITE_OPEN_CREATE
                | OpenFlags::SQLITE_OPEN_FULL_MUTEX,
        )
        .map_err(|error| format!("Cannot open Sonne database: {error}"))?;

        connection
            .busy_timeout(Duration::from_secs(10))
            .map_err(|error| format!("Cannot configure database timeout: {error}"))?;

        connection
            .execute_batch(
                "
                PRAGMA foreign_keys = ON;
                PRAGMA journal_mode = WAL;
                PRAGMA synchronous = NORMAL;
                PRAGMA temp_store = MEMORY;
                PRAGMA mmap_size = 268435456;
                ",
            )
            .map_err(|error| format!("Cannot configure SQLite: {error}"))?;

        let current_version: i64 = connection
            .pragma_query_value(None, "user_version", |row| row.get(0))
            .map_err(|error| format!("Cannot read database version: {error}"))?;

        if current_version > SCHEMA_VERSION {
            return Err(format!(
                "Database version {current_version} is newer than supported version {SCHEMA_VERSION}"
            ));
        }

        if current_version < SCHEMA_VERSION {
            connection
                .execute_batch(INITIAL_SCHEMA)
                .map_err(|error| format!("Cannot initialize database schema: {error}"))?;

            connection
                .pragma_update(None, "user_version", SCHEMA_VERSION)
                .map_err(|error| format!("Cannot update database version: {error}"))?;
        }

        Ok(Self {
            connection,
            root_dir,
        })
    }

    pub fn connection(&self) -> &Connection {
        &self.connection
    }

    pub fn root_dir(&self) -> &Path {
        &self.root_dir
    }
}

use std::{
    fs::File,
    io::{Read, Seek, SeekFrom},
    sync::Mutex,

use tauri::{
    AppHandle, Manager, Runtime, State, UriSchemeResponder,
    http::{Method, Request, Response, StatusCode, header},
};

use super::Storage;

const MAX_RESPONSE_SIZE: u64 = 2 * 1024 * 1024;

pub fn handle_preview_request<R: Runtime>(
    app: AppHandle<R>,
    request: Request<Vec<u8>>,
    responder: UriSchemeResponder,
) {
    let response = match build_preview_response(&app, &request) {
        Ok(response) => response,
        Err(status) => empty_response(status),
    };

    let _ = responder.respond(response);
}

fn build_preview_response<R: Runtime>(
    app: &AppHandle<R>,
    request: &Request<Vec<u8>>,
) -> Result<Response<Vec<u8>>, StatusCode> {
    if request.method() != Method::GET && request.method() != Method::HEAD {
        return Err(StatusCode::METHOD_NOT_ALLOWED);
    }

    let segments: Vec<&str> = request.uri().path().trim_matches('/').split('/').collect();

    if segments.len() != 2
        || segments.iter().any(|segment| {
            segment.is_empty()
                || !segment
                    .bytes()
                    .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
        })
    {
        return Err(StatusCode::BAD_REQUEST);
    }

    let chat_id = segments[0];
    let reference_id = segments[1];

    let state: State<'_, Mutex<Storage>> = app.state();
    let (path, file_name, mime_type, file_size) = {
        let storage = state
            .lock()
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

        let preview = storage
            .get_preview_blob(chat_id, reference_id)
            .map_err(|_| StatusCode::NOT_FOUND)?;

        let path = storage
            .resolve_blob_path(&preview.relative_path)
            .map_err(|_| StatusCode::NOT_FOUND)?;

        (
            path,
            preview.file_name,
            safe_mime_type(&preview.mime_type).to_string(),
            preview.original_size as u64,
        )
    };

    let range_header = request.headers().get(header::RANGE);
    let (start, end, status) = match range_header {
        Some(value) => {
            let value = value
                .to_str()
                .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;
            let (start, end) = parse_range(value, file_size)?;
            (start, end, StatusCode::PARTIAL_CONTENT)
        }
        None => {
            if file_size > MAX_RESPONSE_SIZE {
                return Err(StatusCode::RANGE_NOT_SATISFIABLE);
            }
            (0, file_size.saturating_sub(1), StatusCode::OK)
        }
    };

    let length = if file_size == 0 { 0 } else { end - start + 1 };

    if length > MAX_RESPONSE_SIZE {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let mut body = Vec::new();

    if request.method() == Method::GET && length > 0 {
        let mut file = File::open(path).map_err(|_| StatusCode::NOT_FOUND)?;
        file.seek(SeekFrom::Start(start))
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

        body.resize(length as usize, 0);
        file.read_exact(&mut body)
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
    }

    let disposition = format!("inline; filename=\"{}\"", safe_filename(&file_name));

    let mut builder = Response::builder()
        .status(status)
        .header(header::CONTENT_TYPE, mime_type)
        .header(header::CONTENT_LENGTH, length.to_string())
        .header(header::ACCEPT_RANGES, "bytes")
        .header(header::CONTENT_DISPOSITION, disposition)
        .header("X-Content-Type-Options", "nosniff")
        .header("Content-Security-Policy", "default-src 'none'; sandbox");

    if status == StatusCode::PARTIAL_CONTENT {
        builder = builder.header(
            header::CONTENT_RANGE,
            format!("bytes {start}-{end}/{file_size}"),
        );
    }

    builder
        .body(body)
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)
}

fn parse_range(value: &str, size: u64) -> Result<(u64, u64), StatusCode> {
    if size == 0 {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let value = value
        .strip_prefix("bytes=")
        .ok_or(StatusCode::RANGE_NOT_SATISFIABLE)?;

    if value.contains(',') {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    let (start, end) = value
        .split_once('-')
        .ok_or(StatusCode::RANGE_NOT_SATISFIABLE)?;

    let (start, end) = if start.is_empty() {
        let suffix: u64 = end.parse().map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;

        if suffix == 0 {
            return Err(StatusCode::RANGE_NOT_SATISFIABLE);
        }

        (size.saturating_sub(suffix), size - 1)
    } else {
        let start: u64 = start
            .parse()
            .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?;

        let end = if end.is_empty() {
            size - 1
        } else {
            end.parse::<u64>()
                .map_err(|_| StatusCode::RANGE_NOT_SATISFIABLE)?
                .min(size - 1)
        };

        (start, end)
    };

    if start >= size || end < start || end - start + 1 > MAX_RESPONSE_SIZE {
        return Err(StatusCode::RANGE_NOT_SATISFIABLE);
    }

    Ok((start, end))
}

fn safe_mime_type(mime: &str) -> &str {
    match mime {
        "application/pdf" | "image/png" | "image/jpeg" | "image/webp" | "image/gif"
        | "image/bmp" | "text/plain" | "text/markdown" => mime,
        _ => "application/octet-stream",
    }
}

fn safe_filename(name: &str) -> String {
    name.chars()
        .filter(|character| {
            character.is_ascii_alphanumeric() || matches!(character, '.' | '-' | '_' | ' ')
        })
        .take(120)
        .collect::<String>()
        .replace('"', "")
}

fn empty_response(status: StatusCode) -> Response<Vec<u8>> {
    Response::builder()
        .status(status)
        .header(header::CONTENT_LENGTH, "0")
        .header("X-Content-Type-Options", "nosniff")
        .body(Vec::new())
        .unwrap_or_else(|_| Response::new(Vec::new()))
}

fn now_timestamp() -> Result<i64, String> {
    let duration = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map_err(|error| format!("System clock error: {error}"))?;

    i64::try_from(duration.as_millis()).map_err(|error| format!("Timestamp overflow: {error}"))
}

    pub fn start_turn(
        &mut self,
        chat_id: &str,
        rewind_message_id: Option<&str>,
        user_content: &str,
        prompt_content: &str,
        assistant_content: &str,
        assistant_model: Option<&str>,
        assistant_mode: Option<&str>,
    ) -> Result<StartTurnRecord, String> {
        let timestamp = now_timestamp()?;

        let user_id = Uuid::new_v4().to_string();
        let assistant_id = Uuid::new_v4().to_string();

        let transaction = self
            .connection
            .transaction()
            .map_err(|e| format!("Cannot start turn transaction: {e}"))?;

        let chat_exists: bool = transaction
            .query_row(
                "SELECT EXISTS(SELECT 1 FROM chats WHERE id = ?1)",
                params![chat_id],
                |row| row.get(0),
            )
            .map_err(|e| format!("Cannot verify chat: {e}"))?;

        if !chat_exists {
            return Err(format!("Chat not found: {chat_id}"));
        }

        if let Some(message_id) = rewind_message_id {
            let (target_rowid, role): (i64, String) = transaction
                .query_row(
                    "SELECT rowid, role FROM messages WHERE id = ?1 AND chat_id = ?2",
                    params![message_id, chat_id],
                    |row| Ok((row.get(0)?, row.get(1)?)),
                )
                .map_err(|e| format!("Cannot find message to edit: {e}"))?;

            if role != "user" {
                return Err("Only user messages can be edited".to_string());
            }

            transaction
                .execute(
                    "UPDATE attachment_references
                 SET message_id = NULL, purpose = 'composer'
                 WHERE message_id = ?1 AND chat_id = ?2",
                    params![message_id, chat_id],
                )
                .map_err(|e| format!("Cannot restore edited attachments: {e}"))?;

            transaction
                .execute(
                    "DELETE FROM messages WHERE chat_id = ?1 AND rowid >= ?2",
                    params![chat_id, target_rowid],
                )
                .map_err(|e| format!("Cannot truncate conversation: {e}"))?;
        }

        transaction
            .execute(
                "INSERT INTO messages (
                id, chat_id, role, content, prompt_content,
                model_used, mode_used, created_at
            ) VALUES (?1, ?2, 'user', ?3, ?4, NULL, NULL, ?5)",
                params![user_id, chat_id, user_content, prompt_content, timestamp],
            )
            .map_err(|e| format!("Cannot save user message: {e}"))?;

        transaction
            .execute(
                "UPDATE attachment_references
             SET message_id = ?1, purpose = 'message'
             WHERE chat_id = ?2 AND purpose = 'composer'",
                params![user_id, chat_id],
            )
            .map_err(|e| format!("Cannot promote attachments: {e}"))?;

        transaction
            .execute(
                "INSERT INTO messages (
                id, chat_id, role, content, prompt_content,
                model_used, mode_used, created_at
            ) VALUES (?1, ?2, 'assistant', ?3, NULL, ?4, ?5, ?6)",
                params![
                    assistant_id,
                    chat_id,
                    assistant_content,
                    assistant_model,
                    assistant_mode,
                    timestamp
                ],
            )
            .map_err(|e| format!("Cannot create assistant message: {e}"))?;

        transaction
            .execute(
                "UPDATE chats SET updated_at = ?1 WHERE id = ?2",
                params![timestamp, chat_id],
            )
            .map_err(|e| format!("Cannot update chat timestamp: {e}"))?;

        let attachments = {
            let mut statement = transaction
                .prepare(
                    "SELECT
                    r.id,
                    r.chat_id,
                    r.message_id,
                    r.attachment_version_id,
                    v.attachment_id,
                    v.version_number,
                    v.file_name,
                    v.extension,
                    v.mime_type,
                    v.blob_sha256,
                    b.original_size,
                    r.position,
                    r.purpose,
                    r.created_at
                 FROM attachment_references r
                 JOIN attachment_versions v
                    ON v.id = r.attachment_version_id
                 JOIN blobs b
                    ON b.sha256 = v.blob_sha256
                 WHERE r.message_id = ?1
                 ORDER BY r.position ASC, r.created_at ASC",
                )
                .map_err(|e| format!("Cannot prepare turn attachment query: {e}"))?;

            let rows = statement
                .query_map(params![user_id], |row| {
                    Ok(super::AttachmentReferenceRecord {
                        id: row.get(0)?,
                        chat_id: row.get(1)?,
                        message_id: row.get(2)?,
                        attachment_version_id: row.get(3)?,
                        attachment_id: row.get(4)?,
                        version_number: row.get(5)?,
                        file_name: row.get(6)?,
                        extension: row.get(7)?,
                        mime_type: row.get(8)?,
                        blob_sha256: row.get(9)?,
                        original_size: row.get(10)?,
                        position: row.get(11)?,
                        purpose: row.get(12)?,
                        created_at: row.get(13)?,
                    })
                })
                .map_err(|e| format!("Cannot query turn attachments: {e}"))?;

            rows.collect::<Result<Vec<_>, _>>()
                .map_err(|e| format!("Cannot read turn attachments: {e}"))?
        };

        transaction
            .commit()
            .map_err(|e| format!("Cannot commit turn: {e}"))?;

        Ok(StartTurnRecord {
            user_message: MessageRecord {
                id: user_id,
                chat_id: chat_id.to_owned(),
                role: "user".to_string(),
                content: user_content.to_owned(),
                prompt_content: Some(prompt_content.to_owned()),
                model_used: None,
                mode_used: None,
                created_at: timestamp,
            },
            assistant_message: MessageRecord {
                id: assistant_id,
                chat_id: chat_id.to_owned(),
                role: "assistant".to_string(),
                content: assistant_content.to_owned(),
                prompt_content: None,
                model_used: assistant_model.map(str::to_owned),
                mode_used: assistant_mode.map(str::to_owned),
                created_at: timestamp,
            },
            attachments,
        })
    }

        let transaction = self
            .connection
            .transaction()
            .map_err(|error| format!("Cannot start chat rewind: {error}"))?;

        let (target_rowid, role): (i64, String) = transaction
            .query_row(
                "SELECT rowid, role FROM messages WHERE id = ?1 AND chat_id = ?2",
                params![message_id, chat_id],
                |row| Ok((row.get(0)?, row.get(1)?)),
            )
            .map_err(|error| format!("Cannot find message in chat: {error}"))?;

        if role != "user" {
            return Err("Only user messages can be edited".to_string());
        }

        transaction
            .execute(
                "UPDATE attachment_references
             SET message_id = NULL, purpose = 'composer'
             WHERE message_id = ?1 AND chat_id = ?2",
                params![message_id, chat_id],
            )
            .map_err(|error| format!("Cannot restore message attachments: {error}"))?;

        transaction
            .execute(
                "DELETE FROM messages WHERE chat_id = ?1 AND rowid >= ?2",
                params![chat_id, target_rowid],
            )
            .map_err(|error| format!("Cannot remove later messages: {error}"))?;

        transaction
            .execute(
                "UPDATE chats SET updated_at = ?1 WHERE id = ?2",
                params![timestamp, chat_id],
            )
            .map_err(|error| format!("Cannot update chat timestamp: {error}"))?;

        transaction
            .commit()
            .map_err(|error| format!("Cannot commit chat rewind: {error}"))?;

        Ok(())
    }

        let deleted = self
            .connection
            .execute("DELETE FROM chats WHERE id = ?1", params![chat_id])
            .map_err(|error| format!("Cannot delete chat: {error}"))?;

        Ok(deleted > 0)
    }

/**
 * Removes an embedded <think>...</think> block if a model
 * accidentally includes one in its normal response content.
 *
 * The separate Ollama thinking channel is never displayed.
 */
function extractAnswer(raw: string): string {
  const thinkOpen = raw.indexOf("<think>");

  if (thinkOpen === -1) {
    return raw.trim();
  }

  const thinkClose = raw.indexOf("</think>", thinkOpen);

  if (thinkClose === -1) {
    return raw.slice(0, thinkOpen).trim();
  }

  return (
    raw.slice(0, thinkOpen) + raw.slice(thinkClose + "</think>".length)
  ).trim();
}

const PERF_DIAGNOSTICS_ENABLED = true;

type PerfMarkName =
  | "instructions"
  | "headers"
  | "firstChunk"
  | "firstThinking"
  | "firstContent"
  | "firstVisible"
  | "firstPaint"
  | "doneChunk"
  | "streamEnd";

type PerfTrace = {
  t0: number;
  model: string;
  mode: ResolvedMode;
  modeSetting: ModeOption;
  modelSelection: "auto" | "manual";
  profile: ModelProfile;
  backend: ExecutionBackend;
  thinkingChannel: boolean;
  historyMessages: number;
  historyChars: number;
  context: ContextStats | null; // estimates from contextManager
  systemPromptChars: number;
  userPromptChars: number;
  marks: Partial<Record<PerfMarkName, number>>;
  chunks: number;
  thinkingChunks: number;
  contentChunks: number;
  lineFailures: number;
  streamError: string | null;
  cancelled: boolean;
  final: OllamaChatChunk | null;
};

function perfMark(trace: PerfTrace, name: PerfMarkName): void {
  // Only the first occurrence of each mark is recorded.
  if (trace.marks[name] === undefined) {
    trace.marks[name] = performance.now();
  }
}

function perfObserveChunk(trace: PerfTrace, chunk: OllamaChatChunk): void {
  trace.chunks += 1;

  const hasThinking = Boolean(chunk.message?.thinking);
  const hasContent = Boolean(chunk.message?.content);

  if (hasThinking) {
    trace.thinkingChunks += 1;
    perfMark(trace, "firstThinking");
  }

  if (hasContent) {
    trace.contentChunks += 1;
    perfMark(trace, "firstContent");
  }

  if (hasThinking || hasContent) {
    perfMark(trace, "firstChunk");
  }

  if (typeof chunk.error === "string" && trace.streamError === null) {
    trace.streamError = chunk.error.slice(0, 200);
  }

  if (chunk.done === true) {
    trace.final = chunk;
    perfMark(trace, "doneChunk");
  }
}

function perfIsValidNumber(value: unknown): value is number {
  return typeof value === "number" && Number.isFinite(value) && value >= 0;
}

/** Ollama reports durations in nanoseconds. Absent value -> "N/A". */
function perfNsToMs(value: unknown): string {
  return perfIsValidNumber(value) ? `${(value / 1e6).toFixed(0)} ms` : "N/A";
}

/** Milliseconds elapsed from t0 to a mark. Missing mark -> "N/A". */
function perfSince(trace: PerfTrace, name: PerfMarkName): string {
  const mark = trace.marks[name];

  return mark === undefined ? "N/A" : `${(mark - trace.t0).toFixed(0)} ms`;
}

/** Tokens per second from an Ollama count and a nanosecond duration. */
function perfRate(count: unknown, durationNs: unknown): string {
  if (!perfIsValidNumber(count) || !perfIsValidNumber(durationNs)) {
    return "N/A";
  }

  if (count === 0 || durationNs === 0) return "N/A";

  return `${(count / (durationNs / 1e9)).toFixed(1)} tok/s`;
}

function perfCount(value: unknown): string {
  return perfIsValidNumber(value) ? String(value) : "N/A";
}

function perfDiff(later?: number, earlier?: number): string {
  if (later === undefined || earlier === undefined) return "N/A";

  return `${(later - earlier).toFixed(0)} ms`;
}

// Context Manager metrics. All "est_" values are local estimates; the actual
// prompt size is prompt_tokens_prompt_eval_count (Ollama). Metadata only.
function perfContextReport(ctx: ContextStats | null): Record<string, string> {
  if (!ctx) return { context_optimization: "N/A" };

  return {
    context_optimization: ctx.optimized ? "active" : "not needed",
    context_status: ctx.status,
    context_window: `${ctx.contextWindow} (${ctx.windowSource})`,
    context_input_budget_est: String(ctx.inputBudget),
    context_output_reserve: String(ctx.outputReserve),
    history_messages_total: String(ctx.totalHistoryMessages),
    history_messages_omitted: String(ctx.omittedHistoryMessages),
    est_history_tokens_original: String(ctx.estOriginalHistoryTokens),
    est_history_tokens_retained: String(ctx.estSelectedHistoryTokens),
    est_history_tokens_removed: String(ctx.estRemovedHistoryTokens),
    est_fixed_tokens_system_plus_user: String(ctx.estFixedTokens),
    est_input_tokens_total: String(ctx.estInputTokens),
  };
}

function perfBuildReport(trace: PerfTrace): Record<string, string> {
  const final = trace.final;
  const marks = trace.marks;

  const wallMs =
    marks.streamEnd !== undefined ? marks.streamEnd - trace.t0 : undefined;

  const ollamaTotalMs = perfIsValidNumber(final?.total_duration)
    ? final.total_duration / 1e6
    : undefined;

  return {
    model_requested: trace.model,
    model_reported_by_ollama: final?.model ?? "N/A",
    mode_resolved: trace.mode,
    mode_setting: trace.modeSetting,
    model_selection: trace.modelSelection,
    profile: trace.profile,
    backend: trace.backend,
    thinking_channel_requested: String(trace.thinkingChannel),

    wall_total: wallMs === undefined ? "N/A" : `${wallMs.toFixed(0)} ms`,
    ollama_total_duration: perfNsToMs(final?.total_duration),
    client_overhead_wall_minus_ollama:
      wallMs === undefined || ollamaTotalMs === undefined
        ? "N/A"
        : `${(wallMs - ollamaTotalMs).toFixed(0)} ms`,
    ollama_load_duration: perfNsToMs(final?.load_duration),
    ollama_prompt_eval_duration: perfNsToMs(final?.prompt_eval_duration),
    ollama_eval_duration: perfNsToMs(final?.eval_duration),

    ttft_first_visible_token_queued: perfSince(trace, "firstVisible"),
    ttft_first_visible_token_painted_approx: perfSince(trace, "firstPaint"),
    time_to_first_any_chunk: perfSince(trace, "firstChunk"),
    time_to_first_thinking_chunk: perfSince(trace, "firstThinking"),
    time_to_first_content_chunk: perfSince(trace, "firstContent"),
    thinking_phase_first_thinking_to_first_content: perfDiff(
      marks.firstContent,
      marks.firstThinking,
    ),
    time_to_instructions_ready: perfSince(trace, "instructions"),
    time_to_response_headers: perfSince(trace, "headers"),
    time_to_done_chunk: perfSince(trace, "doneChunk"),

    prompt_tokens_prompt_eval_count: perfCount(final?.prompt_eval_count),
    output_tokens_eval_count: perfCount(final?.eval_count),
    prompt_eval_rate: perfRate(
      final?.prompt_eval_count,
      final?.prompt_eval_duration,
    ),
    generation_rate: perfRate(final?.eval_count, final?.eval_duration),

    done_chunk_received: String(trace.final !== null),
    done_reason: final?.done_reason ?? "N/A",
    stream_error: trace.streamError ?? "none",
    stream_cancelled: String(trace.cancelled),
    stream_completed: String(trace.final !== null && !trace.cancelled),
    stream_chunks_total: String(trace.chunks),
    stream_chunks_thinking: String(trace.thinkingChunks),
    stream_chunks_content: String(trace.contentChunks),
    stream_line_failures: String(trace.lineFailures),

    history_messages_sent: String(trace.historyMessages),
    history_chars_sent: String(trace.historyChars),
    ...perfContextReport(trace.context),
    system_prompt_chars: String(trace.systemPromptChars),
    user_prompt_chars: String(trace.userPromptChars),
  };
}

function perfLogReport(trace: PerfTrace): void {
  if (!PERF_DIAGNOSTICS_ENABLED) return;

  console.info("[Sonne perf] response metrics");
  console.table(perfBuildReport(trace));
  void perfLogOllamaResidency(trace.model);
}

function perfLogFailure(trace: PerfTrace, message: string): void {
  if (!PERF_DIAGNOSTICS_ENABLED) return;

  console.warn("[Sonne perf] request failed", {
    model: trace.model,
    backend: trace.backend,
    mode: trace.mode,
    elapsed: `${(performance.now() - trace.t0).toFixed(0)} ms`,
    time_to_response_headers: perfSince(trace, "headers"),
    error: message.slice(0, 200),
  });
}

/**
 * Read-only check of which models Ollama currently keeps in memory
 * (GET /api/ps). Does not change the keep-alive timer.
 */
async function perfLogOllamaResidency(requestedModel: string): Promise<void> {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 2000);

  try {
    const response = await fetch(`${OLLAMA_URL}/api/ps`, {
      signal: controller.signal,
    });

    if (!response.ok) {
      console.info(`[Sonne perf] /api/ps unavailable: HTTP ${response.status}`);
      return;
    }

    const data = (await response.json()) as {
      models?: Array<{
        name?: string;
        model?: string;
        size?: number;
        size_vram?: number;
        context_length?: number;
        expires_at?: string;
      }>;
    };

    const gib = (bytes: unknown) =>
      perfIsValidNumber(bytes) ? (bytes / 1024 ** 3).toFixed(2) : "N/A";

    const rows = (data.models ?? []).map((entry) => {
      const name = entry.name ?? entry.model ?? "N/A";
      const expires = entry.expires_at ? Date.parse(entry.expires_at) : NaN;

      return {
        model: name,
        is_requested_model: String(name === requestedModel),
        size_gib: gib(entry.size),
        vram_gib: gib(entry.size_vram),
        gpu_share:
          perfIsValidNumber(entry.size) &&
          entry.size > 0 &&
          perfIsValidNumber(entry.size_vram)
            ? `${((entry.size_vram / entry.size) * 100).toFixed(0)}%`
            : "N/A",
        context_length: perfCount(entry.context_length),
        unloads_in: Number.isFinite(expires)
          ? `${Math.max(0, Math.round((expires - Date.now()) / 1000))} s`
          : "N/A",
      };
    });

    console.info(
      `[Sonne perf] models resident in Ollama after response: ${rows.length}`,
    );

    if (rows.length > 0) console.table(rows);
  } catch {
    console.info("[Sonne perf] /api/ps request failed or timed out");
  } finally {
    clearTimeout(timer);
  }
}

function highlightCode(code: string, lang: string): string {
  const normalized =
    lang === "html" || lang === "xml" || lang === "svg"
      ? "markup"
      : lang === "js"
        ? "javascript"
        : lang === "ts"
          ? "typescript"
          : lang === "py"
            ? "python"
            : lang;

  const grammar = Prism?.languages?.[normalized];

  if (grammar && typeof Prism?.highlight === "function") {
    return Prism.highlight(code, grammar, normalized);
  }

  return code
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

export function App() {
  const [models, setModels] = useState<string[]>([]);
  const [modelsLoaded, setModelsLoaded] = useState<boolean>(false);
  const [selectedModel, setSelectedModel] = useState<string>("auto");
  const [modeSetting, setModeSetting] = useState<ModeOption>("auto");
  const [allowThinking, setAllowThinking] = useState<boolean>(false);
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [input, setInput] = useState<string>("");
  const [attachments, setAttachments] = useState<AttachmentReference[]>([]);
  const [chatId, setChatId] = useState<string>("");

  useEffect(() => {
    let active = true;
    const startupSession = chatSessionRef.current;

    async function restoreChat() {
      try {
        const chats = await invoke<Array<{
          id: string;
          title: string;
          createdAt: number;
          updatedAt: number;
        }>>("list_chats");

        if (!active || startupSession !== chatSessionRef.current) return;

        if (chats.length === 0) {
          const chat = await invoke<{ id: string }>("create_chat", {
            title: "Новий чат",
          });
          if (!active || startupSession !== chatSessionRef.current) return;
          setChatId(chat.id);
          return;
        }

        const latestChat = chats[0];
        if (!active || startupSession !== chatSessionRef.current) return;
        setChatId(latestChat.id);

        const [storedMessages, composerAttachments] = await Promise.all([
          invoke<Array<{
            id: string;
            chatId: string;
            role: "user" | "assistant" | "system";
            content: string;
            promptContent: string | null;
            modelUsed: string | null;
            modeUsed: string | null;
            createdAt: number;
          }>>("list_messages", { chatId: latestChat.id }),
          invoke<AttachmentReference[]>("list_composer_attachments", {
            chatId: latestChat.id,
          }),
        ]);

        if (!active || startupSession !== chatSessionRef.current) return;

        const restoredMessages = await Promise.all(
          storedMessages
            .filter((message) => message.role !== "system")
            .map(async (message): Promise<ChatMessage> => {
              const messageAttachments = await invoke<AttachmentReference[]>(
                "list_message_attachments",
                { messageId: message.id },
              );

              return {
                dbId: message.id,
                role: message.role as "user" | "assistant",
                content: message.content,
                promptContent: message.promptContent ?? undefined,
                modelUsed: message.modelUsed ?? undefined,
                modeUsed: (message.modeUsed ?? undefined) as ResolvedMode | undefined,
                attachments: messageAttachments,
                timestamp: new Date(message.createdAt).toLocaleTimeString("uk-UA", {
                  hour: "2-digit",
                  minute: "2-digit",
                  second: "2-digit",
                }),
                includeInHistory: true,
              };
            }),
        );

        if (!active || startupSession !== chatSessionRef.current) return;
        setMessages(restoredMessages);
        setAttachments(composerAttachments);
      } catch (error) {
        console.error("Не вдалося відновити історію чату:", error);
      }
    }

    void restoreChat();

    return () => {
      active = false;
    };
  }, []);
  const [editingMessageIndex, setEditingMessageIndex] = useState<number | null>(null);
  const [isGenerating, setIsGenerating] = useState<boolean>(false);
  const [modelActivity, setModelActivity] = useState<{
    model: string; profile: string; backend: string; thinking: boolean;
  } | null>(null);
  const [elapsedSec, setElapsedSec] = useState<number>(0);
  const [previewHtml, setPreviewHtml] = useState<string | null>(null);

  const bottomRef = useRef<HTMLDivElement | null>(null);
  const sendLockRef = useRef(false);
  const chatSessionRef = useRef(0);
  const activeGenerationRef = useRef<ActiveGeneration | null>(null);
  const generationIdRef = useRef(0);

  // ----------------------------------------------------------
  // LOAD PRISM LANGUAGES
  // ----------------------------------------------------------

  useEffect(() => {
    void Promise.all([
      import("prismjs/components/prism-markup"),
      import("prismjs/components/prism-css"),
      import("prismjs/components/prism-javascript"),
      import("prismjs/components/prism-json"),
      import("prismjs/components/prism-typescript"),
      import("prismjs/components/prism-java"),
      import("prismjs/components/prism-c"),
      import("prismjs/components/prism-python"),
    ]).catch(() => {
      // Some Prism languages may fail to load.
    });
  }, []);

  // ----------------------------------------------------------
  // LOAD OLLAMA MODELS
  // ----------------------------------------------------------

  useEffect(() => {
    async function loadModels() {
      try {
        const response = await fetch(`${OLLAMA_URL}/api/tags`);

        if (!response.ok) return;

        const data = await response.json();

        const names: string[] = (data.models || [])
          .map((model: { name: string }) => model.name)
          .filter((name: string) => isSupportedModelName(name));

        setModels(names);
      } catch {
        // Ollama is offline or unavailable.
      } finally {
        setModelsLoaded(true);
      }
    }

    void loadModels();
  }, []);

  // ----------------------------------------------------------
  // GENERATION TIMER
  // ----------------------------------------------------------

  useEffect(() => {
    if (!isGenerating) return;

    const start = Date.now();

    setElapsedSec(0);

    const timer = setInterval(() => {
      setElapsedSec(Number(((Date.now() - start) / 1000).toFixed(1)));
    }, 100);

    return () => clearInterval(timer);
  }, [isGenerating]);

  // ----------------------------------------------------------
  // AUTO SCROLL
  // ----------------------------------------------------------

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages, isGenerating]);

  // ----------------------------------------------------------
  // PREVIEW
  // ----------------------------------------------------------

  function handleOpenPreview(code: string, lang: string) {
    if (lang === "css") {
      setPreviewHtml(
        `<!DOCTYPE html>
<html>
<head>
<style>${code}</style>
</head>
<body>
<div style="padding:24px;font-family:sans-serif">
CSS підключено
</div>
</body>
</html>`,
      );

      return;
    }

    setPreviewHtml(code);
  }

  function handleStopGeneration() {
    const active = activeGenerationRef.current;
    if (!active) return;

    try {
      active.flush();
    } catch (error) {
      console.error("[Sonne] Помилка фінального flush під час скасування:", error);
    }

    active.cancelled = true;
    active.controller.abort();

    try {
      active.request?.abort();
    } catch (error) {
      console.error("[Sonne] Помилка abort Ollama request:", error);
    }

    setIsGenerating(false);
  }

  async function handleCopy(content: string) {
    try {
      await navigator.clipboard.writeText(content);
    } catch (error) {
      console.error("[Sonne] Не вдалося скопіювати повідомлення:", error);
    }
  }

  function handleEdit(index: number) {
    if (isGenerating) handleStopGeneration();

    const message = messages[index];
    if (!message || message.role !== "user") return;

    const attachmentLabel = message.attachments?.length
      ? `\n\n📎 ${message.attachments.map((file) => file.fileName).join(", ")}`
      : "";

    const editableContent =
      attachmentLabel && message.content.endsWith(attachmentLabel)
        ? message.content.slice(0, -attachmentLabel.length)
        : message.content;

    setInput(editableContent);
    setAttachments([...(message.attachments ?? [])]);
    setEditingMessageIndex(index);
  }

  // ----------------------------------------------------------
  // SEND MESSAGE
  // ----------------------------------------------------------

  async function handleSend() {
    if (sendLockRef.current) return;

    sendLockRef.current = true;

    try {
      await handleSendInternal();
    } finally {
      sendLockRef.current = false;
    }
  }

  async function handleSendInternal() {
    const sendSession = chatSessionRef.current;
    const isEditing = editingMessageIndex !== null;
    const baseMessages = isEditing
      ? messages.slice(0, editingMessageIndex)
      : messages;
    const attachmentFiles = [...attachments];
    const prompt = input.trim() || (attachmentFiles.length
      ? "Проаналізуй прикріплені файли та надай змістовну відповідь."
      : "");

    if (!prompt || isGenerating) return;

    const userTime = formatNow();
    const startTime = Date.now();
    const perfT0 = performance.now(); // [perf-diag]

    const resolvedMode = detectIntent(prompt, modeSetting);
    const requiresModel = resolvedMode !== "image";

    if (requiresModel && !modelsLoaded) return;

    // Profile and execution backend are resolved together. Pro AI has two
    // backends (Thinking / Instruct); a missing backend is an explicit error,
    // never a silent substitution.
    const route = requiresModel
      ? routeRequest(
          resolvedMode,
          models,
          selectedModel,
          prompt,
          modeSetting === "auto",
          allowThinking,
        )
      : null;

    if (route && !route.ok) {
      const unavailableText = route.message;

      if (!chatId) {
        console.error("Чат ще не ініціалізовано.");
        return;
      }

      let rewindMessageId: string | null = null;
      if (isEditing) {
        const editedMessage = messages[editingMessageIndex!];
        if (!editedMessage?.dbId) {
          console.error("Не вдалося визначити повідомлення для редагування.");
          return;
        }
        rewindMessageId = editedMessage.dbId;
      }

      const userContent = prompt + (attachmentFiles.length
        ? `\\n\\n📎 ${attachmentFiles.map((file) => file.fileName).join(", ")}`
        : "");

      const startedTurn = await invoke<{
        userMessage: { id: string };
        assistantMessage: { id: string };
        attachments: AttachmentReference[];
      }>("start_turn", {
        chatId,
        rewindMessageId,
        userContent,
        promptContent: prompt,
        assistantContent: unavailableText,
        assistantModel: null,
        assistantMode: resolvedMode,
      });

      if (sendSession !== chatSessionRef.current) return;
      const savedUser = startedTurn.userMessage;
      const savedAssistant = startedTurn.assistantMessage;
      const promotedAttachments = startedTurn.attachments;

      setMessages([
        ...baseMessages,
        {
          role: "user",
          dbId: savedUser.id,
          content: userContent,
          promptContent: prompt,
          attachments: promotedAttachments,
          timestamp: userTime,
        },
        {
          role: "assistant",
          dbId: savedAssistant.id,
          content: unavailableText,
          modeUsed: resolvedMode,
          timestamp: userTime,
        },
      ]);

      setInput("");
      setAttachments([]);
      setEditingMessageIndex(null);
      return;
    }

    const backendConfig: BackendConfig | null = route && route.ok ? route.backend : null;
    const modelToUse = backendConfig ? backendConfig.ollamaModel : "Image Generator";
    const activeModel = modelToUse;
    const modelProfile = backendConfig ? backendConfig.profile : null;
    if (backendConfig && route && route.ok) {
      setModelActivity({
        model: backendConfig.ollamaModel,
        profile: backendConfig.profile,
        backend: backendConfig.backend,
        thinking: Boolean(route.think),
      });
    }

    if (requiresModel && !backendConfig) return;

    const generationId = requiresModel ? ++generationIdRef.current : undefined;
    const generationController = requiresModel ? new AbortController() : null;

    if (!chatId) {
      console.error("Чат ще не ініціалізовано.");
      return;
    }

    let rewindMessageId: string | null = null;
    if (isEditing) {
      const editedMessage = messages[editingMessageIndex!];
      if (!editedMessage?.dbId) {
        console.error("Не вдалося визначити повідомлення для редагування.");
        return;
      }
      rewindMessageId = editedMessage.dbId;
    }

      // Validate text-generation dependencies before creating the SQLite turn.
      if (
        resolvedMode !== "image" &&
        (!modelProfile || !backendConfig || !generationController || generationId === undefined)
      ) {
        console.error("Не вдалося підготувати текстову генерацію: відсутні параметри моделі.");
        return;
      }

    const userContent = prompt + (attachmentFiles.length
      ? `

📎 ${attachmentFiles.map((file) => file.fileName).join(", ")}`
      : "");

    const imageUnavailableText =
      "Режим генерації зображень поки не підключений. Запит збережено; " +
      "реальну генерацію додамо після інтеграції рушія.";

    const startedTurn = await invoke<{
      userMessage: { id: string };
      assistantMessage: { id: string };
      attachments: AttachmentReference[];
    }>("start_turn", {
      chatId,
      rewindMessageId,
      userContent,
      promptContent: prompt,
      assistantContent: resolvedMode === "image" ? imageUnavailableText : "",
      assistantModel: resolvedMode === "image" ? null : modelToUse,
      assistantMode: resolvedMode,
    });

    if (sendSession !== chatSessionRef.current) return;
    const savedUser = startedTurn.userMessage;
    const savedAssistant = startedTurn.assistantMessage;
    const promotedAttachments = startedTurn.attachments;

    // Add user message and an empty assistant message.
    setMessages([
      ...baseMessages,
      {
        role: "user",
        dbId: savedUser.id,
        content: userContent,
        promptContent: prompt,
        attachments: promotedAttachments,
        timestamp: userTime,
        turnId: generationId,
      },
      {
        id: generationId,
        dbId: savedAssistant.id,
        turnId: generationId,
        role: "assistant",
        content: resolvedMode === "image" ? imageUnavailableText : "",
        modeUsed: resolvedMode,
        modelUsed: resolvedMode === "image" ? undefined : modelToUse,
        timestamp: userTime,
      },
    ]);

    setInput("");
    setAttachments([]);
    setEditingMessageIndex(null);

    // --------------------------------------------------------
    // IMAGE MODE — GENERATOR NOT CONNECTED
    // --------------------------------------------------------

    if (resolvedMode === "image") {
      setMessages((previous) => {
        const updated = [...previous];
        updated[updated.length - 1] = {
          role: "assistant",
          modeUsed: "image",
          timestamp: formatNow(),
          content: imageUnavailableText,
        };
        return updated;
      });
      return;
    }

    if (!modelProfile) return;

    if (!backendConfig) return;

    if (!generationController || generationId === undefined) return;

    setIsGenerating(true);

    // for the chosen backend (Pro Thinking: true, Pro Instruct: false,

    const shouldUseThinkingChannel = route && route.ok ? route.think : false;

    // input is reduced, later, by selectHistory().
    const fullHistory = baseMessages
      .filter((message) => message.includeInHistory === true)
      .map((message) => ({
        role: message.role,
        content: message.content,
      }));

    const perfTrace: PerfTrace = {
      t0: perfT0,
      model: modelToUse,
      mode: resolvedMode,
      modeSetting,
      modelSelection: selectedModel === "auto" ? "auto" : "manual",
      profile: modelProfile,
      backend: backendConfig.backend,
      thinkingChannel: shouldUseThinkingChannel,
      historyMessages: fullHistory.length,
      context: null,
      historyChars: fullHistory.reduce(
        (total, message) => total + message.content.length,
        0,
      ),
      systemPromptChars: 0,
      userPromptChars: prompt.length,
      marks: {},
      chunks: 0,
      thinkingChunks: 0,
      contentChunks: 0,
      lineFailures: 0,
      streamError: null,
      cancelled: false,
      final: null,
    };

    let rawFull = "";
    let pendingContent = "";
    let animationFrame: number | null = null;

    function updateAssistant(update: Partial<ChatMessage>) {
      if (sendSession !== chatSessionRef.current) return;
      setMessages((previous) =>
        previous.map((message) => {
          if (message.id === generationId) {
            return { ...message, ...update };
          }

          if (
            update.includeInHistory === true &&
            message.role === "user" &&
            message.turnId === generationId
          ) {
            return { ...message, includeInHistory: true };
          }

          return message;
        }),
      );
    }

    function flushPendingContent() {
      if (animationFrame !== null) {
        cancelAnimationFrame(animationFrame);
        animationFrame = null;
      }

      if (activeGeneration.cancelled) {
        pendingContent = "";
        return;
      }

      if (!pendingContent) return;

      rawFull += pendingContent;
      pendingContent = "";

      const answer = extractAnswer(rawFull);

      // [perf-diag] First non-empty answer text that reaches the UI.
      if (answer && perfTrace.marks.firstVisible === undefined) {
        perfMark(perfTrace, "firstVisible");
        requestAnimationFrame(() =>
          requestAnimationFrame(() => perfMark(perfTrace, "firstPaint")),
        );
      }

      updateAssistant({
        content: answer,
        modeUsed: resolvedMode,
        modelUsed: activeModel,
        timestamp: formatNow(),
      });
    }

    const activeGeneration: ActiveGeneration = {
      id: generationId,
      controller: generationController,
      request: null,
      cancelled: false,
      flush: flushPendingContent,

      try {
        await invoke("update_message", {
          messageId: savedAssistant.id,
          content,
          promptContent: null,
          modelUsed: modelToUse,
          modeUsed: resolvedMode,
        });
      } catch (error) {
        console.error("[Sonne] Не вдалося зберегти відповідь у SQLite:", error);
      }

    // RUNTIME INSTRUCTIONS + OLLAMA REQUEST
    // --------------------------------------------------------

      const runtimeInstructions = await invoke<string>(
        "get_runtime_instructions",
        { model: modelProfile },
      );

      if (activeGeneration.cancelled || generationController.signal.aborted) {
        throw new DOMException("Generation cancelled", "AbortError");

      // same task/response assembly (see buildSystemPrompt).
      const systemPrompt = buildSystemPrompt(runtimeInstructions, resolvedMode);

      perfTrace.systemPromptChars = systemPrompt.length; // [perf-diag]

      // SQLite transaction is the source of truth: process exactly the
      // attachments promoted into this message by start_turn.
      const attachmentResults = await Promise.all(
        promotedAttachments.map((file) => readStoredAttachment(file)),
      );

      const attachmentContext = attachmentResults.map((file) =>
        `\n\n--- Вкладений файл: ${file.name} (${file.type}) ---\n${file.content}`,
      ).join("");

      const modelPrompt = attachmentContext
        ? `${prompt}\n\nВміст вкладених файлів:${attachmentContext}`
        : prompt;

      const contextBudget = resolveContextBudget(
        backendConfig,
        shouldUseThinkingChannel,
      );

      const fixedTokens =
        estimateMessageTokens({ content: systemPrompt }) +
        estimateMessageTokens({ content: modelPrompt });

      if (fixedTokens > contextBudget.inputBudget) {
        throw new Error(
          `Системний промпт і вкладення потребують приблизно ${fixedTokens} токенів, ` +
          `але доступний бюджет — ${contextBudget.inputBudget}. ` +
          `Розділи документи на менші частини або надішли менше вмісту за раз.`,
        );
      }

      // Smart History: older conversation turns are omitted first.
      const contextSelection = selectHistory(fullHistory, {
        systemPrompt,
        userPrompt: modelPrompt,
        budget: contextBudget,
      });
      const historyMessages = contextSelection.history;

      perfTrace.context = contextSelection.stats; // [perf-diag]
      perfTrace.historyMessages = historyMessages.length; // [perf-diag]
      perfTrace.historyChars = historyMessages.reduce(
        (total, message) => total + message.content.length,
        0,
      ); // [perf-diag]

      if (contextSelection.stats.status === "fixed-content-exceeds-budget") {
        console.warn(
          "[Sonne context] system prompt + current message exceed the estimated input budget; sent without history, not truncated.",
          contextSelection.stats,
        );
      }



      const apiMessages = [
        {
          role: "system" as const,
          content: systemPrompt,
        },
        ...historyMessages,
        {
          role: "user" as const,
          content: modelPrompt,
        },
      ];

      const activeRequest = await createChatStream(
        OLLAMA_URL,
        buildChatRequestBody(
          backendConfig,
          apiMessages,
          userRequestedThinking,
        ),
        generationController.signal,
      );

      perfMark(perfTrace, "headers"); // [perf-diag]

      activeGeneration.request = activeRequest;

      if (activeGeneration.cancelled || generationController.signal.aborted) {
        activeRequest.abort();
        throw new DOMException("Generation cancelled", "AbortError");
      }

      // ------------------------------------------------------
      // STREAM PROCESSING
      // ------------------------------------------------------

      function queueContent(contentChunk: string) {
        if (
          !contentChunk ||
          activeGeneration.cancelled ||
          activeGenerationRef.current?.id !== generationId
        ) {
          return;
        }

        pendingContent += contentChunk;

        if (animationFrame === null) {
          animationFrame = requestAnimationFrame(() => {
            animationFrame = null;
            if (activeGeneration.cancelled) return;
            flushPendingContent();
          });
        }
      }

      for await (const chunk of activeRequest.stream) {
        if (
          activeGeneration.cancelled ||
          generationController.signal.aborted
        ) {
          break;
        }

        perfObserveChunk(perfTrace, chunk); // [perf-diag]

        // The private thinking channel is intentionally ignored.
        // Only normal response content is displayed.
        queueContent(chunk.message?.content || "");
      }

      if (activeGeneration.cancelled || generationController.signal.aborted) {
        throw new DOMException("Generation cancelled", "AbortError");
      }

      flushPendingContent();
      perfMark(perfTrace, "streamEnd"); // [perf-diag]
      perfLogReport(perfTrace); // [perf-diag]

      // ------------------------------------------------------
      // FINISH RESPONSE
      // ------------------------------------------------------

      const totalSec = ((Date.now() - startTime) / 1000).toFixed(1);

      updateAssistant({
        timestamp: formatNow(),
        durationSec: totalSec,
        includeInHistory: true,
      });

      await persistAssistantMessage(extractAnswer(rawFull));
    } catch (err) {
      // ------------------------------------------------------
      // ERROR HANDLING
      // ------------------------------------------------------

      const errMsg =
        err instanceof Error ? err.message : "Помилка з'єднання з Ollama";

      flushPendingContent();

      if (activeGeneration.cancelled || generationController.signal.aborted) {
        perfTrace.cancelled = true; // [perf-diag]
        perfMark(perfTrace, "streamEnd"); // [perf-diag]
        perfLogReport(perfTrace); // [perf-diag]
        updateAssistant({
          timestamp: formatNow(),
          durationSec: ((Date.now() - startTime) / 1000).toFixed(1),
        });
        await persistAssistantMessage(extractAnswer(rawFull));
        return;
      }

      perfLogFailure(perfTrace, errMsg); // [perf-diag]

      updateAssistant({
        role: "assistant",
        modeUsed: resolvedMode,
        modelUsed: modelToUse,
        content: `Не вдалося обробити запит: ${errMsg}`,
        timestamp: formatNow(),
      });

      await persistAssistantMessage(`Не вдалося обробити запит: ${errMsg}`);
    } finally {
      try {
        flushPendingContent();
      } catch (error) {
        console.error("[Sonne] Помилка фінального оновлення UI:", error);
      }

      try {
        activeGeneration.request?.dispose();
      } catch (error) {
        console.error("[Sonne] Помилка очищення Ollama request:", error);
      }

      if (activeGenerationRef.current?.id === generationId) {
        activeGenerationRef.current = null;
        setIsGenerating(false);
      }
    }
  }

  // ==========================================================
  // RENDER
  // ==========================================================

  return (
    <div className={`sonne-shell ${isGenerating ? "is-generating" : ""}`}>
      <div className="ambient-layer" />

      <ChatHeader
        models={models}
        selectedModel={selectedModel}
        onSelectedModelChange={setSelectedModel}
        modeSetting={modeSetting}
        onModeSettingChange={setModeSetting}
        allowThinking={allowThinking}
        onToggleThinking={() => setAllowThinking((value) => !value)}
        messagesLength={messages.length}
        onNewChat={() => {
          chatSessionRef.current += 1;
          handleStopGeneration();
          void (async () => {
            try {
              const chat = await invoke<{ id: string }>("create_chat", {
                title: "Новий чат",
              });

              setChatId(chat.id);
              setMessages([]);
              setInput("");
              setAttachments([]);
              setEditingMessageIndex(null);
              setPreviewHtml(null);
            } catch (error) {
              console.error("Не вдалося створити новий чат:", error);
            }
          })();
        }}
        isGenerating={isGenerating}
        modelActivity={modelActivity}
        elapsedSec={elapsedSec}
      />

      <div className="sonne-workspace">
        <div className="chat-pane">
          <main className="sonne-messages">
            {messages.length === 0 ? (
              <EmptyState />
            ) : (
              messages.map((message, index) => (
                <ChatMessageItem
                  key={index}
                  message={message}
                  index={index}
                  totalMessages={messages.length}
                  isGenerating={isGenerating}
                  onPreview={handleOpenPreview}
                  highlightCode={highlightCode}
                  onCopy={handleCopy}
                  onEdit={handleEdit}
                />
              ))
            )}

            <div ref={bottomRef} />
          </main>

          <ChatInput
            input={input}
            onInputChange={setInput}
            attachments={attachments}
            onAttachmentsChange={setAttachments}
            isGenerating={isGenerating}
            elapsedSec={elapsedSec}
            onSend={handleSend}
            onStop={handleStopGeneration}
          chatId={chatId}
            />
        </div>

        {previewHtml !== null && (
          <LivePreview html={previewHtml} onClose={() => setPreviewHtml(null)} />
        )}
      </div>
    </div>
  );
}

import { invoke } from "@tauri-apps/api/core";
import { getDocument, GlobalWorkerOptions } from "pdfjs-dist";
import mammoth from "mammoth";

GlobalWorkerOptions.workerSrc = new URL(
  "pdfjs-dist/build/pdf.worker.min.mjs",
  import.meta.url,
).toString();

export type SupportedFileType = "text" | "pdf" | "docx";

export type ReadFileResult = {
  name: string;
  type: SupportedFileType;
  size: number;
  content: string;
};

const TEXT_EXTENSIONS = new Set([
  "txt", "md", "json", "csv", "html", "htm", "css",
  "js", "jsx", "ts", "tsx", "py", "java", "c", "cpp",
  "h", "hpp", "cs", "go", "rs", "php", "rb", "swift",
  "kt", "sql", "xml", "yaml", "yml", "toml", "ini",
  "sh", "bash", "zsh", "env", "log", "gitignore",
  "dockerfile",
]);

export function getFileType(file: File): SupportedFileType {
  const extension = file.name.split(".").pop()?.toLowerCase() ?? "";

  if (extension === "pdf") return "pdf";
  if (extension === "docx") return "docx";
  if (TEXT_EXTENSIONS.has(extension)) return "text";

  throw new Error(
    `Непідтримуваний формат файлу: ${extension || "unknown"}`,
  );
}

async function readTextFile(file: File): Promise<string> {
  return file.text();
}

async function readPdfFile(file: File): Promise<string> {
  const data = new Uint8Array(await file.arrayBuffer());
  const pdf = await getDocument({ data }).promise;

  try {
    const pages: string[] = [];

    for (let pageNumber = 1; pageNumber <= pdf.numPages; pageNumber++) {
      const page = await pdf.getPage(pageNumber);
      const textContent = await page.getTextContent();

      const pageText = textContent.items
        .filter((item) => "str" in item)
        .map((item) => ("str" in item ? item.str : ""))
        .join(" ");

      pages.push(
        `--- Сторінка ${pageNumber} ---\n${pageText}`,
      );
    }

    return pages.join("\n\n");
  } finally {
    await pdf.cleanup();
  }
}

async function readDocxFile(file: File): Promise<string> {
  const arrayBuffer = await file.arrayBuffer();
  const result = await mammoth.extractRawText({ arrayBuffer });
  return result.value;
}

export async function readFile(file: File): Promise<ReadFileResult> {
  const type = getFileType(file);
  let content: string;

  switch (type) {
    case "text":
      content = await readTextFile(file);
      break;
    case "pdf":
      content = await readPdfFile(file);
      break;
    case "docx":
      content = await readDocxFile(file);
      break;
  }

  if (!content.trim()) {
    throw new Error(`Не вдалося отримати текст із файлу "${file.name}".`);
  }

  return {
    name: file.name,
    type,
    size: file.size,
    content,
  };
}


export interface StoredAttachmentInput {
  id: string;
  chatId: string;
  fileName: string;
  mimeType: string;
  originalSize: number;
}

export async function readStoredAttachment(
  attachment: StoredAttachmentInput,
): Promise<ReadFileResult> {
  const CHUNK_SIZE = 256 * 1024;
  const chunks: ArrayBuffer[] = [];
  let offset = 0;

  while (offset < attachment.originalSize) {
    const length = Math.min(
      CHUNK_SIZE,
      attachment.originalSize - offset,
    );

    const bytes = await invoke<number[]>("read_attachment_chunk", {
      chatId: attachment.chatId,
      referenceId: attachment.id,
      offset,
      length,
    });

    if (!bytes.length || bytes.length > length) {
      throw new Error(
        `Не вдалося прочитати вкладення "${attachment.fileName}" повністю.`,
      );
    }

    const chunk = Uint8Array.from(bytes);
    chunks.push(chunk.buffer as ArrayBuffer);
    offset += bytes.length;
  }

  if (offset !== attachment.originalSize) {
    throw new Error(
      `Розмір прочитаного файлу "${attachment.fileName}" не збігається з оригіналом.`,
    );
  }

  const file = new File(chunks, attachment.fileName, {
    type: attachment.mimeType || "application/octet-stream",
  });

  return readFile(file);
}

import { useEffect, useRef, useState } from "react";
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";
import { getCurrentWebview } from "@tauri-apps/api/webview";

export interface AttachmentReference {
  id: string;
  chatId: string;
  messageId: string | null;
  attachmentVersionId: string;
  attachmentId: string;
  versionNumber: number;
  fileName: string;
  extension: string;
  mimeType: string;
  blobSha256: string;
  originalSize: number;
  position: number;
  purpose: string;
  createdAt: string;
}

interface FileAttachmentsProps {
  chatId: string;
  attachments: AttachmentReference[];
  onAttachmentsChange: (attachments: AttachmentReference[]) => void;
}

function formatFileSize(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  if (bytes < 1024 * 1024 * 1024) {
    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
  }
  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`;
}

export default function FileAttachments({
  chatId,
  attachments,
  onAttachmentsChange,
}: FileAttachmentsProps) {
  const [isDragging, setIsDragging] = useState(false);
  const [error, setError] = useState("");
  const [isImporting, setIsImporting] = useState(false);
  const importInProgress = useRef(false);

  useEffect(() => {
    let disposed = false;
    let unlistenDrop: (() => void) | undefined;
    let unlistenImported: (() => void) | undefined;
    let unlistenError: (() => void) | undefined;

    async function setupListeners() {
      const importedListener = await listen<AttachmentReference[]>(
        "attachments-imported",
        (event) => {
          if (disposed) return;
          onAttachmentsChange(event.payload);
          setError("");
          setIsImporting(false);
          importInProgress.current = false;
        },
      );

      const errorListener = await listen<string>(
        "attachments-import-error",
        (event) => {
          if (disposed) return;
          setError(event.payload);
          setIsImporting(false);
          importInProgress.current = false;
        },
      );

      const dropListener = await getCurrentWebview().onDragDropEvent(
        (event) => {
          if (disposed) return;

          switch (event.payload.type) {
            case "enter":
            case "over":
              setIsDragging(true);
              break;

            case "leave":
              setIsDragging(false);
              break;

            case "drop": {
              setIsDragging(false);
              const paths = event.payload.paths;

              if (paths.length > 0) {
                void importPaths(paths);
              }
              break;
            }
          }
        },
      );

      if (disposed) {
        importedListener();
        errorListener();
        dropListener();
        return;
      }

      unlistenImported = importedListener;
      unlistenError = errorListener;
      unlistenDrop = dropListener;
    }

    void setupListeners().catch((cause) => {
      if (!disposed) {
        setError(
          cause instanceof Error
            ? cause.message
            : "Не вдалося підключити обробники вкладень.",
        );
      }
    });

    return () => {
      disposed = true;
      unlistenImported?.();
      unlistenError?.();
      unlistenDrop?.();
    };
  }, [chatId, onAttachmentsChange]);

  async function importPaths(paths: string[]) {
    if (importInProgress.current || paths.length === 0) return;

    importInProgress.current = true;
    setIsImporting(true);
    setError("");

    try {
      await invoke("import_files", { chatId, paths });
    } catch (cause) {
      importInProgress.current = false;
      setIsImporting(false);
      setError(
        cause instanceof Error ? cause.message : "Не вдалося імпортувати файли.",
      );
    }
  }

  async function openFileDialog() {
    if (importInProgress.current) return;

    setError("");

    try {
      await invoke("import_attachments", { chatId });
    } catch (cause) {
      setError(
        cause instanceof Error
          ? cause.message
          : "Не вдалося відкрити діалог вибору файлів.",
      );
    }
  }

  async function removeAttachment(referenceId: string) {
    setError("");

    try {
      await invoke("remove_attachment_reference", {
        referenceId,
      });

      const updated = await invoke<AttachmentReference[]>(
        "list_composer_attachments",
        { chatId },
      );

      onAttachmentsChange(updated);
    } catch (cause) {
      setError(
        cause instanceof Error
          ? cause.message
          : "Не вдалося видалити вкладення.",
      );
    }
  }

  return (
    <div className={`file-attachments ${isDragging ? "is-dragging" : ""}`}>
      <button
        type="button"
        className="file-attach-btn"
        onClick={() => void openFileDialog()}
        disabled={isImporting}
        title="Додати файли"
      >
        <span aria-hidden="true">＋</span> Додати файли
      </button>

      {attachments.length > 0 && (
        <div className="file-attachments-list">
          {attachments.map((attachment) => (
            <div className="file-attachment" key={attachment.id}>
              <div className="file-attachment-preview">
                <span className="file-attachment-name">
                  {attachment.fileName}
                </span>
                <span className="file-attachment-size">
                  {formatFileSize(attachment.originalSize)}
                </span>
              </div>

              <button
                type="button"
                className="file-attachment-remove"
                onClick={() => void removeAttachment(attachment.id)}
                aria-label={`Видалити ${attachment.fileName}`}
                title="Видалити файл"
              >
                ×
              </button>
            </div>
          ))}
        </div>
      )}

      {isImporting && (
        <div className="file-attachments-status">Імпорт файлів...</div>
      )}

      {error && (
        <div className="file-attachments-error" role="alert">
          {error}
        </div>
      )}

      {isDragging && (
        <div className="file-drop-overlay">
          Відпусти файли, щоб додати їх
        </div>
      )}
    </div>
  );
}

// ============================================================
// SONNE CONTEXT MANAGER (pure, UI-independent, no imports)
//
// Phase 2A: Context Budget + Smart History.
//
// Decides which part of the conversation is sent to the model. It works
// on a COPY of the history that is about to be sent; the conversation
// shown in the UI and stored by Sonne is never touched.
//
// All token numbers here are LOCAL ESTIMATES, not Ollama token counts.
// The real number is `prompt_eval_count` on Ollama's final chunk.
// ============================================================

export type ContextMessage = { role: "user" | "assistant"; content: string };

/** Structural subset of BackendConfig (modelRouting.ts), so this file needs no imports. */
export type ContextModelConfig = {
  ollamaModel: string;
  options?: { num_ctx: number; num_predict: number };
};

// ------------------------------------------------------------
// CONFIGURATION (single place to maintain)
// ------------------------------------------------------------

/**
 * Context windows (num_ctx) for models whose window is NOT sent per request,
 * i.e. models with a Sonne Modelfile. Keyed by exact Ollama model name.
 * Fill in from each Modelfile's `PARAMETER num_ctx`; do not guess.
 * Models with request-level `options.num_ctx` (Pro Instruct) use that value
 * automatically, so it can never drift from what Ollama is actually given.
 */
export const MODEL_CONTEXT_WINDOWS: Readonly<Record<string, number>> = {
  "sonne-gemma:latest": 8192,
  "sonne-pro-ai:latest": 12288,
};

/** Used for any model without a known window. Equals Ollama's classic default. */
export const FALLBACK_CONTEXT_WINDOW = 4096;

/** Share of the window kept free for the answer (more when the model thinks). */
export const OUTPUT_RESERVE_RATIO = 0.25;
export const OUTPUT_RESERVE_RATIO_THINKING = 0.35;
export const MIN_OUTPUT_RESERVE_TOKENS = 512;

/** Extra headroom because the estimator is approximate. */
export const SAFETY_MARGIN_RATIO = 0.05;

/** Rough per-message cost of role/template tokens. */
export const MESSAGE_OVERHEAD_TOKENS = 4;

// ------------------------------------------------------------
// TOKEN ESTIMATION (estimate only; deliberately on the high side)
// ------------------------------------------------------------

/**
 * Weighted character count, no tokenizer. Roughly: Cyrillic 2 chars/token,
 * Latin ~3 chars/token, punctuation/digits/symbols 2 chars/token, whitespace
 * cheap, other scripts 1 token/char. Not exact; tuned to over- not under-count.
 */
export function estimateTokens(text: string): number {
  let weight = 0;

  for (let i = 0; i < text.length; i++) {
    const c = text.charCodeAt(i);

    if (c === 32 || c === 10 || c === 9 || c === 13) weight += 0.15;
    else if ((c >= 65 && c <= 90) || (c >= 97 && c <= 122)) weight += 0.32;
    else if (c >= 0x0400 && c <= 0x052f) weight += 0.5; // Cyrillic
    else if (c < 128) weight += 0.5; // digits, punctuation, symbols
    else if (c >= 0xd800 && c <= 0xdfff) weight += 0.75; // surrogate half (emoji etc.)
    else weight += 1; // CJK and other scripts

  }

  return Math.ceil(weight);
}

export function estimateMessageTokens(message: { content: string }): number {
  return estimateTokens(message.content) + MESSAGE_OVERHEAD_TOKENS;
}

// ------------------------------------------------------------
// CONTEXT BUDGET
// ------------------------------------------------------------

export type ContextBudget = {
  contextWindow: number;
  windowSource: "request-options" | "model-table" | "fallback";
  outputReserve: number;
  safetyMargin: number;
  /** Max estimated tokens for system prompt + history + current message. */
  inputBudget: number;
};

export function resolveContextBudget(
  config: ContextModelConfig,
  thinking: boolean,
): ContextBudget {
  let contextWindow = FALLBACK_CONTEXT_WINDOW;
  let windowSource: ContextBudget["windowSource"] = "fallback";

  if (config.options && config.options.num_ctx > 0) {
    contextWindow = config.options.num_ctx;
    windowSource = "request-options";
  } else if (MODEL_CONTEXT_WINDOWS[config.ollamaModel] > 0) {
    contextWindow = MODEL_CONTEXT_WINDOWS[config.ollamaModel];
    windowSource = "model-table";
  }

  const ratio = thinking ? OUTPUT_RESERVE_RATIO_THINKING : OUTPUT_RESERVE_RATIO;
  let outputReserve = Math.max(MIN_OUTPUT_RESERVE_TOKENS, Math.floor(contextWindow * ratio));

  // A smaller configured output limit needs no larger reserve.
  if (config.options && config.options.num_predict > 0) {
    outputReserve = Math.min(outputReserve, config.options.num_predict);
  }

  const safetyMargin = Math.ceil(contextWindow * SAFETY_MARGIN_RATIO);
  const inputBudget = Math.max(0, contextWindow - outputReserve - safetyMargin);

  return { contextWindow, windowSource, outputReserve, safetyMargin, inputBudget };
}

// ------------------------------------------------------------
// SMART HISTORY (recent-first, whole turns, deterministic)
// ------------------------------------------------------------

export type HistorySelectionStatus =
  | "within-budget" // everything fits; history unchanged
  | "trimmed" // oldest turns omitted
  | "fixed-content-exceeds-budget"; // system prompt + current message alone exceed the budget

export type ContextStats = {
  status: HistorySelectionStatus;
  /** True when any history was omitted from the model input. */
  optimized: boolean;
  contextWindow: number;
  windowSource: ContextBudget["windowSource"];
  inputBudget: number;
  outputReserve: number;
  // All token values below are ESTIMATES (not Ollama's prompt_eval_count).
  estFixedTokens: number; // system prompt + current user message
  estOriginalHistoryTokens: number;
  estSelectedHistoryTokens: number;
  estRemovedHistoryTokens: number;
  estInputTokens: number; // fixed + selected history
  totalHistoryMessages: number;
  selectedHistoryMessages: number;
  omittedHistoryMessages: number;
};

export type HistorySelection = {
  /** New array; the input array and its message objects are never modified. */
  history: ContextMessage[];
  stats: ContextStats;
};

export function selectHistory(
  history: readonly ContextMessage[],
  input: { systemPrompt: string; userPrompt: string; budget: ContextBudget },
): HistorySelection {
  const { budget } = input;
  const costs = history.map(estimateMessageTokens);
  const originalTokens = costs.reduce((sum, cost) => sum + cost, 0);
  const fixedTokens =
    estimateMessageTokens({ content: input.systemPrompt }) +
    estimateMessageTokens({ content: input.userPrompt });
  const available = budget.inputBudget - fixedTokens;

  let keepFrom = 0;
  let status: HistorySelectionStatus = "within-budget";

  if (available < 0) {
    // System prompt and current message are never cut. Send them as they are
    // without history and let the caller surface the overflow.
    keepFrom = history.length;
    status = "fixed-content-exceeds-budget";
  } else if (originalTokens > available) {
    status = "trimmed";
    keepFrom = history.length;

    // Walk back one turn at a time (a user message plus the replies that
    // follow it). A turn that does not fit stops the walk, so what is kept
    // is always a contiguous, whole-turn tail of the conversation.
    let used = 0;
    let end = history.length;

    while (end > 0) {
      let start = end - 1;

      while (start > 0 && history[start].role !== "user") start--;

      // Replies with no user message before them would be orphaned.
      if (history[start].role !== "user") break;

      let turnCost = 0;

      for (let i = start; i < end; i++) turnCost += costs[i];
      if (used + turnCost > available) break;

      used += turnCost;
      keepFrom = start;
      end = start;
    }
  }

  const selected = history.slice(keepFrom).map((message) => ({
    role: message.role,
    content: message.content,
  }));
  const selectedTokens = costs.slice(keepFrom).reduce((sum, cost) => sum + cost, 0);
  const omitted = history.length - selected.length;

  return {
    history: selected,
    stats: {
      status,
      optimized: omitted > 0,
      contextWindow: budget.contextWindow,
      windowSource: budget.windowSource,
      inputBudget: budget.inputBudget,
      outputReserve: budget.outputReserve,
      estFixedTokens: fixedTokens,
      estOriginalHistoryTokens: originalTokens,
      estSelectedHistoryTokens: selectedTokens,
      estRemovedHistoryTokens: originalTokens - selectedTokens,
      estInputTokens: fixedTokens + selectedTokens,
      totalHistoryMessages: history.length,
      selectedHistoryMessages: selected.length,
      omittedHistoryMessages: omitted,
    },
  };
}

export function containsTerm(text: string, term: string): boolean {
  const escaped = term.replace(/[.*+?^\${}()|[\]\\]/g, "\\$&");
  const pattern = new RegExp(
    `(^|[^\\p{L}\\p{N}_])${escaped}(?=$|[^\\p{L}\\p{N}_])`,
    "iu",
  );

  return pattern.test(text);
}

export function detectIntent(
  prompt: string,
  modeSetting: ModeOption,
): ResolvedMode {
  if (modeSetting !== "auto") return modeSetting;

  const q = prompt.toLowerCase().trim();

  const imageRequests = [
    "намалюй",
    "згенеруй картинку",
    "згенеруй зображення",
    "зроби картинку",
    "зроби зображення",
    "draw",
    "generate image",
    "create image",
    "image of",
  ];

  if (imageRequests.some((request) => containsTerm(q, request))) {
    return "image";
  }

  // Translation intent is recognized before formatting, code-term and
  // complexity checks: a technical translation is still a translation.
  if (isTranslationRequest(q)) return "chat";

  const formattingRequests = [
    "markdown",
    "inline code",
    "жирний текст",
    "курсив",
    "таблиця",
    "таблицею",
    "форматування",
  ];

  if (formattingRequests.some((request) => containsTerm(q, request))) {
    return "chat";
  }

  const task = classifyTask(q).type;

  const explainsCode =
    task === "explanation" &&
    /(?:цей|цей\s+самий|наведен\p{L}*)\s+код|код(?:у)?\s+(?:на|мовою)|(?:this|the|following)\s+(?:code|snippet)|code\s+snippet/iu.test(q);

  if (
    task === "coding" ||
    task === "debugging" ||
    explainsCode
  ) {
    return "code";
  }

  if (isComplexTechnicalRequest(q)) {
    return "code";
  }

  return "chat";
}

const WORD_END = "(?=$|[^\\p{L}\\p{N}_])";

/**

/**
 * Plain technical explanations (no code terms) that should use Pro AI
 * Instruct in Auto mode. Translation requests are excluded on purpose so
 * they keep their existing Gemma behavior.
 */
export function isTechnicalExplanationRequest(prompt: string): boolean {
  const q = prompt.toLowerCase().trim();

  if (isTranslationRequest(q)) return false;

  const clauses = q.split(/(?<=[.!?;])\s+|\n+/u);

  return clauses.some((clause) => {
    const asksForExplanation =
      /(?:^|[^\p{L}\p{N}_])(?:explain|explanation|describe|how does|how do|how is|what is|what are|why does|why is|поясни\p{L}*|розкажи\p{L}*|як працює|як працюють|що таке|чому|яка різниця|чим\p{L}*.{0,80}?відрізняється)(?=$|[^\p{L}\p{N}_])/u.test(clause);

    return asksForExplanation &&
      TECHNICAL_TERMS.some((term) => containsTerm(clause, term));
  });
}

/**
 * Translation requests are never re-routed by the complexity classifier.
 *
 * A translation word makes the message a translation request unless the
 * user is asking to BUILD software that does translation. That is the case
 * when the software artifact itself ("function", "script", ...) of an
 * authoring phrase comes before the translation word ("Write a function
 * that performs translation"), or is the noun the translation word directly
 * modifies ("Write a translation function"). An artifact mentioned only
 * after the translation word ("Write a paragraph about translation and
 * include a function example") or text quoted after a leading translation
 * instruction ("Translate: write a function") stays a translation request.
 */
export function isTranslationRequest(prompt: string): boolean {
  const q = prompt.toLowerCase().trim();
  const translation = TRANSLATION_PATTERN.exec(q);

  if (!translation) return false;

  const authoring = SOFTWARE_AUTHORING_PATTERN.exec(q);
  const artifact = authoring?.groups?.artifact;

  if (!authoring || !artifact) return true;

  const artifactEnd = authoring.index + authoring[0].length;
  const artifactStart = artifactEnd - artifact.length;
  const translationEnd = translation.index + translation[0].length;

  // Artifact named before the translation word.
  if (artifactEnd <= translation.index) return false;

  // "translation function": the translation word directly modifies the artifact.
  if (artifactStart >= translationEnd && q.slice(translationEnd, artifactStart).trim() === "") {
    return false;
  }

  return true;
}

// ------------------------------------------------------------
// COMPLEXITY CLASSIFIER (decides Pro AI Thinking vs Instruct)
//
// Strategy: a request is complex when it asks for a demanding KIND of work
// (design / analysis / large-scale engineering) AND that work touches
// several system-level aspects (scope, multiplicity, cross-cutting
// concerns). Signals are grouped by meaning; each group counts at most
// once, so repeating words cannot inflate the score and a single
// technical word ("API", "database", "architecture") never decides alone.
// Matching is prefix-based so Ukrainian/English word forms are covered.
// ------------------------------------------------------------

type SignalGroup = "intent" | "scope" | "multiplicity" | "concern";

/** Regex sources; matched at a word start, any word ending. */
const SIGNALS: Record<SignalGroup, readonly string[]> = {

/** Marks a deliberately small request; vetoes borderline scores only. */
const SIMPLICITY_MARKERS = [
  "simple", "basic", "small", "short", "brief", "quick", "tiny", "eli5",
  "простими\\s+слов", "простий", "проста", "просте", "просту", "простою",
  "базов", "невелик", "короткий", "коротко", "швидк", "маленьк",
];

/** Distinct signal groups required (an intent group is always required). */
const COMPLEXITY_THRESHOLD = 3;
/** A simplicity marker only vetoes scores below this. */
const SIMPLICITY_VETO_BELOW = 4;

function compileStartPattern(sources: readonly string[]): RegExp {
  return new RegExp(`(^|[^\\p{L}\\p{N}_])(?:${sources.join("|")})`, "iu");
}

const GROUP_PATTERNS: Record<SignalGroup, RegExp> = {
  intent: compileStartPattern(SIGNALS.intent),
  scope: compileStartPattern(SIGNALS.scope),
  multiplicity: compileStartPattern(SIGNALS.multiplicity),
  concern: compileStartPattern(SIGNALS.concern),
};
const SIMPLICITY_PATTERN = compileStartPattern(SIMPLICITY_MARKERS);

function normalizeForClassifier(prompt: string): string {
  // Typographic apostrophes (U+2019, U+02BC, U+2018) -> ASCII, so the same word matches in either form.
  return prompt.toLowerCase().replace(/[\u2019\u02bc\u2018`]/g, "'");
}

export type ComplexityAssessment = {
  complex: boolean;
  /** Which signal groups matched (each counts once). */
  groups: SignalGroup[];
  score: number;
  reason: "exact-phrase" | "signals" | "translation" | "simple-marker" | "below-threshold";
};

/**
 * Explains why a request was (not) treated as complex. Pure; used by
 * isComplexRequest() and by tests.
 */
export function assessComplexity(prompt: string): ComplexityAssessment {
  // Translation has priority over everything below, including the legacy
  // exact phrases: translating text about architecture is not engineering.
  if (isTranslationRequest(prompt)) {
    return { complex: false, groups: [], score: 0, reason: "translation" };
  }

  // Existing exact phrases always count (outside translation requests).
  if (isComplexTechnicalRequest(prompt)) {
    return { complex: true, groups: [], score: COMPLEXITY_THRESHOLD, reason: "exact-phrase" };
  }

  const q = normalizeForClassifier(prompt);

  const groups = (Object.keys(GROUP_PATTERNS) as SignalGroup[]).filter((group) =>
    GROUP_PATTERNS[group].test(q),
  );
  const score = groups.length;

  if (!groups.includes("intent") || score < COMPLEXITY_THRESHOLD) {
    return { complex: false, groups, score, reason: "below-threshold" };
  }

  if (score < SIMPLICITY_VETO_BELOW && SIMPLICITY_PATTERN.test(q)) {
    return { complex: false, groups, score, reason: "simple-marker" };
  }

  return { complex: true, groups, score, reason: "signals" };
}

/** Pro AI complexity decision: exact phrases OR the signal-based classifier. */
export function isComplexRequest(prompt: string): boolean {
  return assessComplexity(prompt).complex;
}

export type TaskType =
  | "conversation"
  | "explanation"
  | "coding"
  | "analysis"
  | "planning"
  | "translation"
  | "debugging"
  | "comparison"
  | "writing";

export type TaskClassification = {
  type: TaskType;
};

export function classifyTask(prompt: string): TaskClassification {
  const q = prompt.toLowerCase().trim();

  if (isTranslationRequest(q)) return { type: "translation" };

  if (
    /(?:виправ|налагод|знайди помилк|debug|fix the bug|troubleshoot)/iu.test(q) ||
    /\bfix\b.{0,60}\b(?:css|html|javascript|typescript|react|python|sql|code|код)\b/iu.test(q)
  ) {
    return { type: "debugging" };
  }

  if (
    /(?:порівняй|порівняти|compare|contrast|відмінност)/iu.test(q)
  ) {
    return { type: "comparison" };
  }

  if (
    /(?:напиши|створи|реалізуй|розроби|write|create|implement|build|develop)/iu.test(q) &&
    /(?:функці|function|код|code|скрипт|script|компонент|component|модул|module|endpoint|ендпоїнт|api)/iu.test(q)
  ) {
    return { type: "coding" };
  }

  if (
    /(?:оптимізуй|оптимізувати|перепиши|рефактор|optimize|optimise|refactor)/iu.test(q) &&
    /(?:sql|javascript|typescript|react|python|html|css|код|code|query|запит|function|функці|component|компонент|api|endpoint)/iu.test(q)
  ) {
    return { type: "coding" };
  }

  if (
    /(?:напиши|створи|склади|сформулюй|write|draft|compose)/iu.test(q) &&
    /(?:лист|повідомлення|статт|опис|есе|cover letter|email|message|article|post)/iu.test(q)
  ) {
    return { type: "writing" };
  }

  if (
    /(?:спроєктуй|спроектуй|проєктуй|проектуй|сплануй|спланувати|design|architect|plan the system)/iu.test(q) &&
    /(?:api|архітектур|architecture|system|систем|database|баз[а-яіїєґ]* даних|backend|бекенд|інфраструктур|infrastructure|pipeline|пайплайн|компонент|component|модул|module)/iu.test(q)
  ) {
    return { type: "planning" };
  }

  if (
    /(?:проаналізуй|аналіз|досліди|оцінити|evaluate|analy[sz]e|investigate)/iu.test(q)
  ) {
    return { type: "analysis" };
  }

  if (
    /(?:поясни|пояснити|як працює|що таке|розкажи про|explain|how does|what is)/iu.test(q) &&
    (
      isTechnicalExplanationRequest(q) ||
      /(?:javascript|typescript|react|python|html|css|sql|api|docker|websocket|(?:програмн|системн|мікросервісн|розподілен)[а-яіїєґ]*\s+архітектур|архітектур[а-яіїєґ]*\s+(?:мікросервіс|розподілен|системн|програмн)|(?:software|system|microservice|distributed)\s+architecture)/iu.test(q)
    )
  ) {
    return { type: "explanation" };
  }

  return { type: "conversation" };
}

export type RoutingErrorCode =
  | "backend-unavailable"
  | "no-supported-models";

export type RouteResult =
  | {
      ok: true;
      backend: BackendConfig;
      /**
       * What the user's Thinking toggle means for this request. Only Gemma
       * consumes it (Pro AI backends decide `think` themselves); it keeps the
       * existing Gemma rule: the toggle applies in Code mode only.
       */
      userRequestedThinking: boolean;
      /** Final `think` value for /api/chat, resolved for the chosen backend. */
      think: boolean;
    }
  | {
      ok: false;
      code: RoutingErrorCode;
      backend: ExecutionBackend;
      message: string;
    };

/**
 * Existing Gemma thinking rule (moved from App.tsx): the toggle only
 * takes effect in Code mode.
 */
export function resolveUserRequestedThinking(
  resolvedMode: ResolvedMode,
  allowThinking: boolean,
): boolean {
  return resolvedMode === "code" && allowThinking;
}

/**
 * Pro AI backend. The Thinking toggle overrides complexity:
 * - allowThinking ON  -> always Pro AI Thinking.
 * - allowThinking OFF -> by task complexity (complex -> Thinking, else Instruct).
 * Never falls back to the other backend when a model is missing.
 */
export function chooseProBackend(
  prompt: string,
  allowThinking: boolean,
): ExecutionBackend {
  if (allowThinking) return "pro-thinking";

  return isComplexRequest(prompt) ? "pro-thinking" : "pro-instruct";
}

/**
 * Step 1: which profile handles the request. The Thinking toggle is not an
 * input here, so it can never turn a Gemma request into a Pro AI request.
 * Manual model selection always decides the profile.
 */
export function wantedProfile(
  resolvedMode: ResolvedMode,
  selectedModel: string,
  prompt: string,
  isAutoMode: boolean,
): ModelProfile {
  const manual = selectedModel !== "auto" ? getBackendByModelName(selectedModel) : null;

  if (manual) return manual.profile;

  // Explicit Code mode keeps its Pro AI routing.
  if (resolvedMode === "code") return "pro-ai";

  // Auto mode only: genuinely complex requests use Pro AI even without code or
  // technical-explanation keywords. Translations are never complex
  // (see assessComplexity), so they stay on Gemma.
  if (isAutoMode && isComplexRequest(prompt)) return "pro-ai";

  // Auto mode only: plain technical explanations use Pro AI.
  if (isAutoMode && isTechnicalExplanationRequest(prompt)) return "pro-ai";

  return "gemma";
}

/**
 * Step 2: execution backend. Pro AI applies the Thinking override;
 * Gemma is unaffected by it.
 */
export function wantedBackend(
  resolvedMode: ResolvedMode,
  selectedModel: string,
  prompt: string,
  isAutoMode: boolean,
  allowThinking: boolean,
): ExecutionBackend {
  const profile = wantedProfile(resolvedMode, selectedModel, prompt, isAutoMode);

  return profile === "pro-ai" ? chooseProBackend(prompt, allowThinking) : "gemma";
}

/**
 * Router 2.0: task-aware backend selection.
 * Kept separate from wantedBackend() until V2 behavior is verified.
 */
export function wantedBackendV2(
  resolvedMode: ResolvedMode,
  selectedModel: string,
  prompt: string,
  isAutoMode: boolean,
  allowThinking: boolean,
): ExecutionBackend {
  const manual = selectedModel !== "auto"
    ? getBackendByModelName(selectedModel)
    : null;

  // Manual model selection always keeps priority.
  if (manual?.profile === "gemma") return "gemma";

  if (manual?.profile === "pro-ai") {
    return chooseProBackend(prompt, allowThinking);
  }

  // Explicit Code mode keeps its existing Pro AI behavior.
  // Auto-detected Code must not be treated as an explicit selection.
  if (!isAutoMode && resolvedMode === "code") {
    return chooseProBackend(prompt, allowThinking);
  }

  // Explicit non-auto modes preserve their existing behavior.
  if (!isAutoMode) return "gemma";

  // Preserve established Auto-mode technical routing.
  if (
    isComplexRequest(prompt) ||
    isTechnicalExplanationRequest(prompt)
  ) {
    return chooseProBackend(prompt, allowThinking);
  }

  // Router 2.0 task-aware automatic routing.
  const task = classifyTask(prompt).type;

  const proTasks: TaskType[] = [
    "explanation",
    "coding",
    "debugging",
    "analysis",
    "planning",
    "comparison",
  ];

  if (!proTasks.includes(task)) return "gemma";

  return chooseProBackend(prompt, allowThinking);
}

export function unavailableMessage(backend: ExecutionBackend): string {
  const config = getBackendConfig(backend);

  return (
    `Модель для «${config.label}» недоступна в Ollama: ${config.ollamaModel}. ` +
    `Sonne не підміняє її іншою моделлю автоматично.`
  );
}

export function routeRequest(
  resolvedMode: ResolvedMode,
  installedModels: readonly string[],
  selectedModel: string,
  prompt: string,
  isAutoMode: boolean,
  allowThinking: boolean,
): RouteResult {
  const backend = wantedBackendV2(
    resolvedMode,
    selectedModel,
    prompt,
    isAutoMode,
    allowThinking,
  );
  const config = getBackendConfig(backend);

  if (!installedModels.includes(config.ollamaModel)) {
    return {
      ok: false,
      code: installedModels.length === 0 ? "no-supported-models" : "backend-unavailable",
      backend,
      message: unavailableMessage(backend),
    };
  }

  const userRequestedThinking = resolveUserRequestedThinking(resolvedMode, allowThinking);

  return {
    ok: true,
    backend: config,
    userRequestedThinking,
    think: resolveThink(config, userRequestedThinking),
  };
}

  model: string;
  stream: true;
  think: boolean;
  messages: ChatMessageBody[];

};

export function buildChatRequestBody(
  config: BackendConfig,
  messages: ChatMessageBody[],
  userRequestedThinking: boolean,
